Understand the Problem

People visited the Donuts shop by group, if this group of people eat new bach of donuts then
there are happy, other wise(they eat the dounts left by last group) they are unhappy.

how to slove the problem

the initive thoughs is that try every possible permutations all input groups. by the time complexity will be O(2^n)
there is a typical way to reduce the time complexity of this problem, we can group the groups by bathSize, suppose the batch size is k , every group has n/k element, the complexity is O(k^(n/k))

two pre compute method

1 for every group the num can be mod by batsize, it does not influnce the result
2 if the mod is zero we do not need to process this group add them to result directly.

Solution1 simulation TLE

public int maxHappyGroups(int batchSize, int[] groups) {
        int[] count = new int[batchSize];
        for(int x : groups){
            count[x % batchSize]++;
        }
        return dfs(count, 0, 0, groups.length);
    }
    public int dfs(int[] count, int preSum, int i, int n) {
        if (i == n) return 0; // attention there the n is groups.length  not batch size, because we need to process all groups 
        int ret = 0;
        int bonus = preSum % count.length == 0 ? 1 : 0;
        for (int j = 0; j < count.length; j++) {
            if (count[j] == 0) continue;
            count[j]--;
            ret = Math.max(ret, dfs(count, preSum + j, i + 1, n));
            count[j]++;
        }
        return ret + bonus;
    }

First of all, we only need remainder of each group size because rest all in a group will be served a freshly baked batch and they willl have no problem, only the remainder people will be left.

So, for a given state, we explore all the possible cases of collaboration with any group. The DP parameters are int[] arr, where arr[i] denotes the number of groups with i people remaining, and left signifying the number of leftover donuts from the previous batch.

If left is 0, then any group we take will be satisfied because all the persons in that group will be served from freshly baked batches else no matter which group we take, they wont be satisfied.

class Solution {
    HashMap<String,Integer> memo;
    int dfs(int[] arr,int left,int n)
    {
        int q=0;
        String key=Arrays.toString(arr);
        key+=Integer.toString(left);
        if(memo.containsKey(key))
            return memo.get(key);
        else if(left==0)
        {
            for(int i=1;i<n;++i)
                if(arr[i]>0)
                {
                    arr[i]--;
                    q=Math.max(q,1+dfs(arr,n-i,n));
                    arr[i]++;
                }
        }
        else
            for(int i=1;i<n;++i)
                if(arr[i]>0)
                {
                    arr[i]--;
                    q=Math.max(q,dfs(arr,i<=left?left-i:n+left-i,n));
                    arr[i]++;
                }
        memo.put(key,q);
        return q;
        
    }
    public int maxHappyGroups(int n, int[] groups) 
    {
        int arr[]=new int[n];
        for(int x:groups)
            arr[x%n]++;
        memo=new HashMap<String,Integer>();
        return arr[0]+dfs(arr,0,n);
    }
}

class Trie
{
    Trie[] children;
    int val;
    public Trie(int limit)
    {
        children=new Trie[limit];
        val=-1;
    }
    public void insert(Trie root,int[] key,int q,int limit)
    {
        Trie temp=root;
        for(int x:key)
        {
            if(temp.children[x]==null)
                temp.children[x]=new Trie(limit);
            temp=temp.children[x];
        }
        temp.val=q;
    }
    public int search(Trie root,int[] key)
    {
        Trie temp=root;
        for(int x:key)
        {
            if(temp.children[x]==null)
                return -1;
            temp=temp.children[x];
        }
        return temp.val;
    }
}
class Solution {
    Trie memo;
    int dfs(int[] arr,int left,int n,int limit)
    {
        int q=0;
        int key=memo.search(memo,arr) ;
        if(key!=-1)
            return key;
        else if(left==0)
        {

                    arr[i]++;
                }

        int arr[]=new int[n];
        int limit=0;
        for(int x:groups)
            limit=Math.max(limit,++arr[x%n]);
        //memo=new HashMap<String,Integer>();
        memo=new Trie(limit+1);
        return arr[0]+dfs(arr,0,n,limit+1);
    }

class Solution {
    private HashMap<String, Integer> memo = new HashMap<>();
    private int batchSize;
    public int maxHappyGroups(int batchSize, int[] groups) {
        this.batchSize = batchSize;
        int[] counts = new int[batchSize];

        for (int group : groups) {
            counts[group % batchSize]++;
        }

        int result = counts[0];
        counts[0] = 0;

        for (int i = 1; i <= batchSize / 2; i++) {
            if (i * 2 == batchSize) {
                result = result + counts[i] / 2;
                counts[i] %= 2;
            }
            else {
                int pairs = Math.min(counts[i], counts[batchSize - i]);
                result = result + pairs;
                counts[i] = counts[i] - pairs;
                counts[batchSize - i] = counts[batchSize - i] - pairs;
            }
        }
        return result + dfs(counts, 0);

    }

    private int dfs(int[] counts, int leftover) {
        int totalGroups = 0;
        for (int count : counts) {
            totalGroups = totalGroups + count;
        }
        if (totalGroups == 0) return 0;

        String key = Arrays.toString(counts) + "," + leftover;

        if (memo.containsKey(key)) {
            return memo.get(key);
        }
        int maxHappy = 0;
        for (int remainder = 1; remainder < batchSize; remainder++) {
            if (counts[remainder] > 0) {
                int[] newCounts = counts.clone();
                newCounts[remainder]--;
                int newLeftover = (leftover + remainder) % batchSize;

                int happy = (leftover == 0) ? 1 : 0;
                maxHappy = Math.max(maxHappy, happy + dfs(newCounts, newLeftover));
            }
        }

        memo.put(key, maxHappy);
        return maxHappy;
    }
}
