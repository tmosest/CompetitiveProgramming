Intuition

The intuition of the solution is that we can break down the problem of counting the number of special permutations into smaller subproblems. Each subproblem corresponds to a state, where the state is represented by a bitmask that indicates which numbers have already been used.

We can solve each subproblem recursively. The base case is when the state is all zeros, which corresponds to the case where no numbers have been used. In this case, there is only one special permutation, which is the empty permutation.

For the recursive case, we iterate over all the numbers that have not been used yet. For each number, we check if it is divisible by any of the numbers that have already been used. If it is, then we can add the number of special permutations for the next state to the total.

The total number of special permutations is the sum of the number of special permutations for all the states.

  • Time complexity: O(N * (2 ^ N) * N)

  • Space complexity: O(N * (2 ^ N) * N)

C++ Code

class Solution {
public:
  const int mod = 1e9 + 7; 

  int helper(vector<int>& nums, int j, int i, int msk, vector<vector<int>>& dp) {
    // Recursively calculates the number of special permutations of the given numbers.

    if (i == nums.size()) {
      // Base case: If we have reached the end of the numbers, there is only one special permutation.
      return 1;
    }

    if (dp[j + 1][msk] != -3) {
      // Memoization: If we have already calculated the number of special permutations for this state, return it.
      return dp[j + 1][msk];
    }

    int total = mod; // The total number of special permutations.
    for (int ind = 0; ind < nums.size(); ind++) {
      // Iterate over all the numbers.
      if (msk & (1 << ind)) {
        continue; // Skip this number if it has already been used.
      }

      if (j == -1 || nums[ind] % nums[j] == 0 || nums[j] % nums[ind] == 0) {
        // If this number is divisible by the previous number, or if there is no previous number,
        // then add the number of special permutations for the next state to the total.
        total += helper(nums, ind, i + 1, msk | (1 << ind), dp) % mod;
        total %= mod;
      }
    }

    // Save the number of special permutations for this state in the memoization table.
    dp[j + 1][msk] = total;

    // Return the number of special permutations.
    return total;
  }

  int specialPerm(vector<int>& nums) {
    vector<vector<int>> dp(21, vector<int>((1 << 14) + 7, -3));
    return helper(nums, -1, 0, 0, dp) % mod;
  }
};

Java Code

class Solution {
    int MOD = (int)1e9 + 7;
    int N;
    Integer[][] memo;
    int[] nums;
    public int specialPerm(int[] nums) {
        this.N = nums.length;
        this.memo = new Integer[N][1 << N];
        this.nums = nums;
        return backtrack(0, 0);
    }
    //mask = available element to choose from
    private int backtrack(int preIndex, int mask) {
        if (mask == (1 << N) - 1) return 1;
        if (memo[preIndex][mask] != null) return memo[preIndex][mask];
        int count = 0;
        for (int i = 0; i < N; i++)
            if ((mask & (1 << i)) == 0 && 
                (mask == 0 || nums[i] % nums[preIndex] == 0 || nums[preIndex] % nums[i] == 0))
                count = (count + backtrack(i, mask | (1 << i))) % MOD; //used.add(i);
        return memo[preIndex][mask] = count;
    }
}

class Solution:
    MOD = int(1e9) + 7

    def helper(self, nums, j, i, msk, dp):
        if i == len(nums):
            return 1

        if dp[j + 1][msk] != -3:
            return dp[j + 1][msk]

        total = self.MOD
        for ind in range(len(nums)):
            if msk & (1 << ind):
                continue

            if j == -1 or nums[ind] % nums[j] == 0 or nums[j] % nums[ind] == 0:
                total += self.helper(nums, ind, i + 1, msk | (1 << ind), dp) % self.MOD
                total %= self.MOD

        dp[j + 1][msk] = total
        return total

    def specialPerm(self, nums):
        dp = [[-3] * ((1 << 14) + 7) for _ in range(21)]
        return self.helper(nums, -1, 0, 0, dp) % self.MOD

def specialPerm(self, nums: List[int]) -> int:
    n, MOD = len(nums), 10**9 + 7
    @cache
    def dfs(prev, mask):
        if mask == (1 << n) - 1: return 1
        count = 0
        for i in range(n):
            if not (mask & (1 << i)) and (nums[i] % prev == 0 or prev % nums[i] == 0):
                count += dfs(nums[i], mask | (1 << i))
        return count % MOD
    return dfs(1, 0)

int specialPerm(vector<int>& nums) {
    int n = nums.size(), MOD = 1000000007;
    unordered_map<long long, int> cache;

    function<int(int, int)> dfs = [&](int prev, int mask) {
        if (mask == (1 << n) - 1) return 1;
        long long key = (long long)prev << 20 | mask;
        if (cache.count(key)) return cache[key];

        int count = 0;
        for (int i = 0; i < n; i++)
            if (!(mask & (1 << i)) && (nums[i] % prev == 0 || prev % nums[i] == 0))
                count = (count + dfs(nums[i], mask | (1 << i))) % MOD;
        return cache[key] = count;
    };

    return dfs(1, 0);
}

Intuition

Since we need to form a permutation we can start from no elements and then each time we take a element we can see what further element can be taken to form permuatation with it

Approach

Since we are exploring all possible permutation we will apply Dynamic Programming and BitMasking (for keeping visited track)

We will maintain a previous element so that at each point when we take a new element which was previously not taken we can satisfy the condition nums[i] % nums[i + 1] == 0 || nums[i + 1] % nums[i] == 0

To iterate over those elements which are previously not taken we will maintain mask.

Complexity

- Time complexity:
The overall time complexity of the code is O(N * 2^N * N)

- Space complexity:
Since we are using the dp table space complexity goes up to O(14 + 1)(2 ^ 14)

Code (C++)

class Solution {
public:
    int mod = 1e9 + 7;
    int sb(int x)
    {
        int ans = 0;
        while (x > 0)
        {
            if (x & 1)
                ans++;
            x = x >> 1;
        }
        return ans;
    }
    long long dp[1 << 14][15];
    int func(int idx, vector<int> &nums, int mask, int prev)
    {
        if (idx == nums.size())
            return (sb(mask) == nums.size());
        long long way1 = 0;
        if (dp[mask][prev + 1] != -1)
            return dp[mask][prev + 1];
        if (prev == -1)
        {
            for (int i = 0; i < nums.size(); ++i)
            {
                if (mask & (1 << i))
                    continue;
                else
                {
                    int newMask = mask | (1 << i);
                    way1 =(way1 + func(idx + 1, nums, newMask, i)) % mod;
                }
            }
        }
        else
        {
            for (int i = 0; i < nums.size(); ++i)
            {
                if (mask & (1 << i))
                    continue;
                else
                {
                    if (nums[i] % nums[prev] == 0 || nums[prev] % nums[i] == 0)
                    {
                        int newMask = mask | (1 << i);
                        way1 = (way1 + func(idx + 1, nums, newMask, i)) % mod;
                    }
                }
            }
        }
        return dp[mask][prev + 1] = (way1) % mod;
    }
    int specialPerm(vector<int>& nums) {
        memset(dp, -1, sizeof(dp));
        int mask = 0;
        return func(0, nums, mask, -1) % mod;
    }
};

        if (idx == nums.length)
            return (sb(mask) == nums.length) ? 1 : 0;

        long way1 = 0;

        if (dp[mask][prev + 1] != -1)
            return (int) dp[mask][prev + 1];

        if (prev == -1) {
            for (int i = 0; i < nums.length; ++i) {

                    continue;
                else {
                    int newMask = mask | (1 << i);

class Solution:
    mod = 10**9 + 7

    def sb(self, x):
        ans = 0
        while x > 0:
            if x & 1:
                ans += 1
            x = x >> 1
        return ans

    def __init__(self):
        self.dp = [[-1] * 15 for _ in range(1 << 14)]

    def func(self, idx, nums, mask, prev):
        if idx == len(nums):
            return 1 if self.sb(mask) == len(nums) else 0

        way1 = 0

        if self.dp[mask][prev + 1] != -1:
            return self.dp[mask][prev + 1]

        if prev == -1:
            for i in range(len(nums)):
                if mask & (1 << i):
                    continue
                else:
                    newMask = mask | (1 << i)
                    way1 = (way1 + self.func(idx + 1, nums, newMask, i)) % self.mod
        else:
            for i in range(len(nums)):
                if mask & (1 << i):
                    continue
                else:
                    if nums[i] % nums[prev] == 0 or nums[prev] % nums[i] == 0:
                        newMask = mask | (1 << i)
                        way1 = (way1 + self.func(idx + 1, nums, newMask, i)) % self.mod

        self.dp[mask][prev + 1] = way1
        return self.dp[mask][prev + 1]

    def specialPerm(self, nums):
        mask = 0
        return self.func(0, nums, mask, -1) % self.mod

class Solution {
    public int specialPerm(int[] nums) {
        int n = nums.length;
        int[][][] dp =new int[n+1][n][1<<n];
         for (int i = 0; i <=n ; i++) {
            for (int j = 0; j < n; j++) {
                Arrays.fill(dp[i][j], -1);
            }

        }
        return dfs(0, 0, 0, dp, nums);
    }

    private int dfs(int curr, int prev, int mask, int[][][] dp, int[]  nums) {
        if(curr == nums.length){
            return 1; // if the index is reaches the end than this is an valid permutation so we return 1
        }
        if (dp[prev][curr][mask] != -1) { // the value for the prev index and current index and mask is already computed return tat ans i.e no matter what happens if the current and previous and mask is equal the balance permutation is always constant
            return dp[prev][curr][mask];
        }
        int res = 0;
        for (int i = 0; i < nums.length; i++) {
            if ((mask & 1 << i) == 0){ // current number nums[i] is not choosen
                if (curr == 0  || nums[prev] % nums[i] == 0 || nums[i] % nums[prev] == 0) { // current number if(i==0) or nums[i] % prev ==0 or nums[i] % prev % nums[i] ==0. 
                    res +=  dfs(curr + 1, i, mask | (1 << i), dp, nums); // mask | (1 << i) mark the bit i as masked(choosed)
                    res %= 1000000007; // to avoid overflow of interger
                }
            }
        }
        return dp[prev][curr][mask] = res;
    }
}
