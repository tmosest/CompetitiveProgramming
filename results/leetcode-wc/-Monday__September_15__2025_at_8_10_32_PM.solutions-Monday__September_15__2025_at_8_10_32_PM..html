Intuition

  • We remove a bar means its add the squar of prev and next line/vertical.
  • When we remove n consecutive bars then the size of the connected small squar width/height becom n+1.
  • So we need to check for max continuos bars length we can remove from horizontal and max consecutive bars we can remove from vertical.

Approach

  • Sort the vBar and hBar
  • Check the max gap in horizontal and max gap in vertical
  • Take min of the (max horizontal gap , max vertical gap) and return its square as answer.

C++

class Solution {
  int getMaxGap(vector<int> bars){
      sort(bars.begin(), bars.end());
      int count = 2, res = 2;
      for(int i = 1; i < bars.size(); ++i) {
          count = (bars[i-1]+1 == bars[i])? count + 1: 2;
          res = max(res, count);
      }
      return res;
  }

  int maximizeSquareHoleArea(int n, int m, vector<int>& hBars, vector<int>& vBars) {
      int gap = min(getMaxGap(hBars), getMaxGap(vBars));
      return gap * gap;
  }
};

Approach

The code finds the maximum area of a square-shaped hole in a grid with removable horizontal and vertical bars. It sorts the bars, calculates the consecutive lengths of horizontal (mh) and vertical (mv) bars, and finds the minimum of (mh+1) and (mv+1). The result is the square of this minimum, representing the maximum area after removing some bars. The algorithm has a time complexity of O(nlog(n) + mlog(m)), where n and m are the lengths of the horizontal and vertical bars, respectively.

Complexity

  • Time complexity:
O(N*log(N))
  • Space complexity:
O(1)

class Solution {
public:
    int maximizeSquareHoleArea(int n, int m, vector<int>& hBars, vector<int>& vBars) {
        vector<int>h;
        vector<int>v;
        sort(hBars.begin(),hBars.end());
        sort(vBars.begin(),vBars.end());
        for(auto i:hBars)
            h.push_back(i);
        for(auto i:vBars)
            v.push_back(i);
        int mv=1,mh=1;
        int c=1;
        for(int i=1;i<h.size();i++)
        {   
            
            if(h[i]==h[i-1]+1)
                c++;
            else
                c=1;
            mh=max(mh,c);
        }
        c=1;
        for(int i=1;i<v.size();i++)
        {   
            if(v[i]==v[i-1]+1)
                c++;
            else
                c=1;
            mv=max(mv,c);
        }
        int x=min(mh+1,mv+1);
        return x*x;
    }
};

Approaches

(Also explained in the code)

  1. Sorting: Both vectors A and B are sorted to identify consecutive elements easily.

  2. Horizontal and Vertical Counts: The code counts the consecutive elements in vectors A and B horizontally and vertically, respectively, using two loops.

  3. Maximum Consecutive Counts: Variables mxh and mxv store the maximum consecutive counts for horizontal and vertical sequences, respectively.

  4. Maximum Possible Area: The maximum possible area of the square-shaped hole is calculated by taking the minimum of mxh and mxv (as the sides of the square) and incrementing it by 1.

Complexity

  • Time complexity:
    O(NlogN)

  • Space complexity:
    O(N+M)

class Solution {
public:
    // Function to maximize the area of the square-shaped hole
    int maximizeSquareHoleArea(int n, int m, vector<int>& A, vector<int>& B) {
        int N = A.size(); // Number of elements in A vector
        int M = B.size(); // Number of elements in B vector
        
        // Sorting the vectors to identify consecutive elements
        sort(A.begin(), A.end());
        sort(B.begin(), B.end());
        
        int id = 0, mxh = 1, mxv = 1; // Initialization of variables
        
        // Loop through vector A to find the maximum horizontal consecutive count
        for(int i = 0; i < N; ++i) {
            int cnt = 1; // Counter for consecutive elements
            int j = i + 1;
            // Count consecutive elements in A
            while(i + 1 < N && A[i] + 1 == A[i + 1]) {
                ++i;
                ++cnt;
            }
            mxh = max(mxh, cnt); // Update maximum horizontal consecutive count
        }
        
        // Loop through vector B to find the maximum vertical consecutive count
        for(int i = 0; i < M; i++) {
            int cnt = 1; // Counter for consecutive elements
            // Count consecutive elements in B
            while(i + 1 < M && B[i] + 1 == B[i + 1]) {
                ++i;
                ++cnt;
            }
            mxv = max(mxv, cnt); // Update maximum vertical consecutive count
        }
        
        // Calculate the maximum possible area for the square hole
        int ans = (min(mxh, mxv) + 1) * (min(mxh, mxv) + 1);
        return ans; // Return the maximum area of the square-shaped hole
    }
};

Intuition

We need to find the number of consecutive elements in hBars and vBars in order to maximise the area. Then,
# holes = # removed bars (# consecutive elements) + 1

Approach

Complexity

  • Time complexity: O(nlogn), n - length of hBars/vBars
  • Space complexity: 0(n)

Code

class Solution {
    public int maximizeSquareHoleArea(int n, int m, int[] hBars, int[] vBars) {
        int x = find(hBars), y = find(vBars);
        int res = Math.min(x, y) + 1;
        
        return res * res;
    }
    
    private int find(int[] arr) {
        Arrays.sort(arr);
        int res = 1, i = 0, n = arr.length;
        
        while (i < n) {
            int count = 1;
            
            while (i+1 < n && arr[i]+1 == arr[i+1]) {
                i++;
                count++;
            }
            i++;
            res = Math.max(res, count);
        }
        
        return res;
    }
}
