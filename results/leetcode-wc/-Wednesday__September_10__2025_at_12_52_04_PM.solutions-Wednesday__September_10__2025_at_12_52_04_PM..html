Intuition

The problem involves traversing a tree-like structure and determining if a certain condition is met based on the traversal path. One approach to solve this problem is to perform a Depth First Search (DFS) traversal of the tree and check the condition at each step.

Approach

  1. Create a HashMap t to store the parent-child relationships based on the parent array.
  2. Define a recursive function dfs that traverses the tree from the root node (-1) and accumulates the string representation of the path.
  3. Check if the concatenated string from the path is a palindrome. If it is, mark the current node as a valid result
  4. Create a result array of boolean values based on the results map filled during the traversal.

Complexity

  • Time complexity: O(n)
  • Space complexity: O(n)

class Solution:
    def findAnswer(self, parent: List[int], s: str) -> List[bool]:
        t = {}
        results = {}

        for i in range(len(parent)):
            p = parent[i]
            t[p] = t.get(p, []) + [i]

        def dfs(x):
            if x not in t:
                results[x] = True
                return s[x]
            st = ""
            for v in t[x]:
                st += dfs(v)
            st += s[x]
            if st[::-1] == st:
                results[x] = True
            return st

        dfs(-1)

        ans = [False] * len(parent)
        for i in results.keys():
            ans[i] = results[i]
        return ans

class Solution {
    int[] inT;
    int[] outT;
    int time;
    char[] dfsStr;
    int idx;
    int[][] nodeStr;
    ArrayList<Integer>[] al;

    public boolean[] findAnswer(int[] parent, String s) {
        int len = s.length();
        al = new ArrayList[len];
        boolean[] ans = new boolean[len];
        nodeStr = new int[len][2];
        dfsStr = new char[len];
        inT = new int[len];
        outT = new int[len];
        for (int i = 0; i < len; i++)
            al[i] = new ArrayList<>();
        for (int i = 1; i < len; i++) {
            al[parent[i]].add(i);
        }
        dfs(parent, s, 0);
        // System.out.println("in=" + Arrays.toString(inT));
        // System.out.println("out=" + Arrays.toString(outT));
        // System.out.println("disc=" + Arrays.toString(discoveryTime));
        int[] palinLen = countSubstrings(new String(dfsStr));
        // System.out.println(new String(dfsStr));
        // System.out.println("palinLen=" + Arrays.toString(palinLen));
        // for each index I know how long is palindrom around that
        for (int i = 0; i < len; i++) {
            int st = nodeStr[i][0];
            int end = nodeStr[i][1];
            int l = end - st + 1;
            // find its center
            // st = 1 , end = 5 -> 1 2 3 4 5 -> st + l/2
            // st = 1 , end = 4 -> 1 2 3 4 ->
            int centre = st + l / 2;
            int c1 = 2 * (centre + 1) - 1;
            int c2 = 2 * (centre + 1) - 2;
            end = 2 * (end + 1) - 1;
            // System.out.println("i = " + i + " st " + st + " end " + end + " centre " +
            // centre + " c1=" + c1 + " c2=" + c2);
            if (l % 2 == 0)
                ans[i] = (end < c2 + palinLen[c2]);
            else
                ans[i] = (end < c1 + palinLen[c1]);
        }
        return ans;
    }

    void dfs(int[] parent, String s, int node) {
        inT[node] = time++;
        nodeStr[node][0] = idx; // start index
        // String dfsStr = "";
        for (int v : al[node])
            dfs(parent, s, v);

        outT[node] = time++;
        nodeStr[node][1] = idx; // end index
        dfsStr[idx++] = s.charAt(node);

    }

    public int[] countSubstrings(String s) {
        int n = s.length();
        StringBuilder t = new StringBuilder("$").append("#");
        for (char c : s.toCharArray()) {
            t.append(c).append("#");
        }
        t.append("^");
        n = t.length();
        int[] p = new int[n];
        int l = 1, r = 1;
        for (int i = 2; i < t.length() - 1; i++) {
            p[i] = Math.max(0, Math.min(r - i, p[l + (r - i)]));
            while (t.charAt(i - p[i]) == t.charAt(i + p[i])) {
                p[i]++;
            }
            if (i + p[i] > r) {
                l = i - p[i];
                r = i + p[i];
            }
        }
        // System.out.println("t=" + t);
        // System.out.println("p=" + Arrays.toString(p));
        return Arrays.copyOfRange(p, 1, p.length - 1);
    }
}
