class Solution {
     static public int longestPalindrome(String s, String t) {
        char[] arr1 = s.toCharArray();
        char[] arr2 = t.toCharArray();
        int n1 = arr1.length, n2 = arr2.length;
        max = 0;

        int[][] dp = new int[n1][n1];
        int[] LCS1 = longestPalindromeSubstring(arr1, dp, n1);

        dp = new int[n2][n2];
        char[] temp = Arrays.copyOf(arr2, n2);

        reverse(temp);
        int[] LCS2 = longestPalindromeSubstring(temp, dp, n2);
        reverse(LCS2);
        
        Integer[][][] dp2 = new Integer[n1][n2][2];
        int ans = solve(0, n2 - 1, 0, arr1, arr2, LCS1, LCS2, dp2);
        return Math.max(ans, max);
    }

    static int max = 0;

    static int solve(int i, int j, int started, char[] arr1, char[] arr2, int[] LCS1, int[] LCS2, Integer[][][] dp) {
        if (i == arr1.length || j < 0) {
            int a = 0;
            if (i < arr1.length) a = LCS1[i];
            if (j >= 0) a = Math.max(a, LCS2[j]);
            return a;
        }
        if (dp[i][j][started] != null) return dp[i][j][started];

        int a = 0, b = 0, c = 0;
        if (started == 0) {
            b = solve(i, j - 1, 0, arr1, arr2, LCS1, LCS2, dp);
            a = solve(i + 1, j, 0, arr1, arr2, LCS1, LCS2, dp);
        }

        if (arr1[i] == arr2[j]) {
            c = 2 + solve(i + 1, j - 1, 1, arr1, arr2, LCS1, LCS2, dp);
        } else {
            c = Math.max(LCS1[i], LCS2[j]);
        }
        return dp[i][j][started] = Math.max(a, Math.max(c, b));
    }

    static int[] longestPalindromeSubstring(char[] arr, int[][] dp, int n) {
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (arr[i] == arr[j]) dp[i][j] = 1;
            }
        }

        for (int i = n - 2; i >= 0; i--) {
            for (int j = n - 1; j >= 1; j--) {
                if (dp[i][j] == 1) dp[i][j] += dp[i + 1][j - 1];
            }
        }

        int[] LCS1 = new int[n];
        for (int i = 0; i < n; i++) {
            for (int j = i, x = 0; j < n; j++, x++) {
                if (Math.abs(dp[i][j] - dp[i + x][j - x]) == x) {
                    LCS1[i] = Math.max(LCS1[i], x + 1);
                    max = Math.max(max, x + 1);
                }
            }
        }
        return LCS1;
    }

    static void reverse(int[] arr) {
        int i = 0, j = arr.length - 1;
        while (i < j) {
            int swap = arr[i];
            arr[i] = arr[j];
            arr[j] = swap;
            i++;
            j--;
        }
    }

        static public int longestPalindrome(String s, String t) {
        char[] arr1 = s.toCharArray();
        char[] arr2 = t.toCharArray();
        int n1 = arr1.length, n2 = arr2.length;
        max = 0;

        int[][] dp = new int[n1][n1];
        int[] LCS1 = longestPalindromeSubstring(arr1, dp, n1);

        dp = new int[n2][n2];
        char[] temp = Arrays.copyOf(arr2, n2);

        reverse(temp);
        int[] LCS2 = longestPalindromeSubstring(temp, dp, n2);
        reverse(LCS2);

        int[][][] dp2 = new int[n1][n2][2];
        for (int i = n1 - 1; i >= 0; i--) {
            for (int j = 0; j < n2; j++) {
                for (int started = 0; started < 2; started++) {
                    int a = 0, b = 0, c = 2;
                    if (started == 0) {
                        if (j - 1 < 0) b = LCS1[i];
                        else b = dp2[i][j - 1][0];
                        if (i + 1 >= n1) a = LCS2[j];
                        else a = dp2[i + 1][j][0];
                    }

                    if (arr1[i] == arr2[j]) {
                        if (i + 1 >= n1 || j - 1 < 0) {
                            int tempRes = 0;
                            if (i + 1< n1) tempRes = LCS1[i + 1];
                            if (j - 1>= 0) tempRes = Math.max(tempRes, LCS2[j - 1]);

                        }

                        c = Math.max(LCS1[i], LCS2[j]);
                    }
                    dp2[i][j][started] = Math.max(a, Math.max(c, b));
                }
            }
        }

        return dp2[0][n2 - 1][0];

Intuition

Approach

  1. Finding Palindromes within Each String:

    • longestPalSub(string s) uses the "center expansion" technique to iterate through each character (and the gap between characters) as potential centers of a palindrome.
    • It computes the longest palindrome contained entirely within s. This method is also applied to t in the overall solution.

  2. Preprocessing for Expanding Palindromes:

    • The function palinStart(string s) constructs an array lps where lps[i] represents the length of the longest palindrome that starts at index i in s.
    • Similarly, palinEnd(string s) calculates the longest palindrome ending at each index, used for t in the solution.

  3. Identifying Symmetry Between Two Strings:

    • In the longestPalindrome(string s, string t) method, it first determines the maximum palindrome length found within s and t separately.
    • Then, it creates a reversed version of t, named rT.
    • A dynamic programming (DP) table of size (n+1)*(m+1) is used to identify common substrings between s and rT. A common substring of length L corresponds to one half of a palindromic sequence, so the candidate palindrome length is initially 2*L.

  4. Extending the Palindrome with Non-Symmetric Parts:

    • If there is additional string remaining in s after the symmetric part, the precomputed array lpsStart is used to potentially extend the palindrome by adding the longest palindrome starting from the next index.
    • Similarly, for t (using the corresponding index from rT), lpsEnd is used to extend the palindrome on the other side.

  5. Choosing the Best Result:

    • After evaluating all possibilities (palindromes entirely in s or t, as well as those formed by symmetric combinations between s and t), the maximum palindrome length found is returned.

Complexity

  • Time complexity: O(n2)

  • Space complexity: O(n)

Intuition

Check out the "longest palindrome substring" problem's dp solution first. https://leetcode.com/problems/longest-palindromic-substring/ The solution uses the same logic. Using those dp arrays, we compute the longest palindrome lengths for each index, starting at each index.

We'll use the logic that X+palindrome+X is also a palindrome where X is any non-empty string.

We'll also compute the common substrings between s and reversed t.

Then, the length of the longest palindrome is equal to "length of common substring" * 2 + longest palindrome starting after the common part ends. We'll pick longest palindrome in s and tReversed, so for each common text in s and tReversed we'll check 2 possibilities: longest palindrome starting at the ending point of the common string in s, and longest palindrome starting at the ending point of the common string in tReversed.

"common part" + "longer palindrome" + "common part" is also a palindrome.

Finding the common parts in s and tReversed is a little bit cumbersome but the solution is really simple otherwise.

Finally, we also check maximal palindrome lengths in individual strings, in case one of the substrings selected is an empty string. Like in the test case m="b", n="aaaa"

import java.util.*;

class Solution {
    public int longestPalindrome(String s, String t) {
        String tReversed = new StringBuilder(t).reverse().toString();

        List<int[]> sameSections = new ArrayList<>();
        for (int i = 0; i < s.length(); i++) {
            int k = i;
            for (int j = 0; j < tReversed.length(); j++) {
                if (k < s.length() && tReversed.charAt(j) == s.charAt(k)) k++;
                else {
                    if (k > i) sameSections.add(new int[]{i, j - (k - i), k - i});
                    k = i;
                    if (tReversed.charAt(j) == s.charAt(k)) k++;
                }
            }
            if (k > i) sameSections.add(new int[]{i, tReversed.length() - (k - i), k - i});
        }

        boolean[][] dpS = new boolean[s.length()][s.length()];
        boolean[][] dpT = new boolean[tReversed.length()][tReversed.length()];
        fillPalindromeDp(dpS, s);
        fillPalindromeDp(dpT, tReversed);
        int[] longestPalS = new int[s.length()];
        int[] longestPalT = new int[tReversed.length()];

        for (int i = 0; i < dpS.length; i++) {
            if (dpS[i][dpS.length - 1]) longestPalS[i] = dpS.length - i;
            else {
                for (int j = i; j < dpS.length; j++) {
                    if (dpS[i][j]) {
                        longestPalS[i] = j - i + 1;
                    }
                }
            }
        }

        for (int i = 0; i < dpT.length; i++) {
            if (dpT[i][dpT.length - 1]) longestPalT[i] = dpT.length - i;
            else {
                for (int j = i; j < dpT.length; j++) {
                    if (dpT[i][j]) {
                        longestPalT[i] = j - i + 1;
                    }
                }
            }
        }

        int ans = 0;
        for (int[] section : sameSections) {
            int start1 = section[0];
            int start2 = section[1];
            int len = section[2];

            int lenPalAfterS = start1 + len >= longestPalS.length ? 0 : longestPalS[start1 + len];
            int lenPalAfterT = start2 + len >= longestPalT.length ? 0 : longestPalT[start2 + len];

            ans = Math.max(ans, Math.max(lenPalAfterS, lenPalAfterT) + len * 2);
        }

        for (int i = 0; i < longestPalS.length; i++) ans = Math.max(ans, longestPalS[i]);
        for (int i = 0; i < longestPalT.length; i++) ans = Math.max(ans, longestPalT[i]);

        return ans;
    }

    void fillPalindromeDp(boolean[][] dp, String s) {
        for (int i = 0; i < s.length(); i++) dp[i][i] = true;
        for (int i = 0; i < s.length() - 1; i++) {
            if (s.charAt(i) == s.charAt(i + 1)) dp[i][i + 1] = true;
        }

        for (int diff = 2; diff < s.length(); diff++) {
            for (int i = 0; i + diff < s.length(); i++) {
                if (s.charAt(i) == s.charAt(i + diff) && dp[i + 1][i + diff - 1]) {
                    dp[i][i + diff] = true;
                }
            }
        }
    }
}
