class Solution {
public:
    int answer = -1;
    void dfs(int node, vector<int>& edges, unordered_map<int, int>& dist, vector<bool>& visit) {
        visit[node] = true;
        int neighbor = edges[node];

        if (neighbor != -1 && !visit[neighbor]) {
            dist[neighbor] = dist[node] + 1;
            dfs(neighbor, edges, dist, visit);
        } else if (neighbor != -1 && dist.count(neighbor)) {
            answer = max(answer, dist[node] - dist[neighbor] + 1);
        }
    }

    int longestCycle(vector<int>& edges) {
        int n = edges.size();
        vector<bool> visit(n);

        for (int i = 0; i < n; i++) {
            if (!visit[i]) {
                unordered_map<int, int> dist;
                dist[i] = 1;
                dfs(i, edges, dist, visit);
            }
        }
        return answer;
    }
};

Intuition :

  • Return the length of the longest cycle in the graph. If no cycle exists, return -1.

Approach : Depth First Search

  • Here we are using a modified depth-first search algorithm to visit each node in the graph.
  • We keep track of the time at which each node is visited and store it in an array called timeVisited.
  • If a node is visited again, we skip it because it has already been processed.
  • If the current node is visited again, and the time at which it was first visited is greater than or equal to the start time of the current traversal, it means that there is a cycle in the graph that includes the current node.
  • The length of the cycle is the difference between the current time and the time at which the node was first visited.
  • We also keep track of the maximum cycle length found so far and return it as the result.

Complexity :

  • Time complexity : O(n)
  • Space complexity : O(n)

class Solution {
  public int longestCycle(int[] edges) {
    int ans = -1; // Initialize the answer to -1
    int time = 1; // Initialize the current time step to 1
    int[] timeVisited = new int[edges.length]; // Initialize an array to store the time at which each node was first visited

    // Iterate through each node in the graph
    for (int i = 0; i < edges.length; ++i) {
      if (timeVisited[i] > 0) // If the node has already been visited, skip it
        continue;
      final int startTime = time; // Record the start time of the current traversal
      int u = i; // Initialize the current node to the ith node
      // Traverse the graph until the end of the path is reached or a visited node is encountered
      while (u != -1 && timeVisited[u] == 0) {
        timeVisited[u] = time++; // Record the current time step and increment time
        u = edges[u]; // Move to the next node in the path
      }
      // If a cycle is found that includes the current node, update the answer
      if (u != -1 && timeVisited[u] >= startTime)
        ans = Math.max(ans, time - timeVisited[u]);
    }

    return ans; // Return the length of the longest cycle found
  }
}

public int longestCycle(int[] edges) {
    int res = -1;
    boolean[] vis = new boolean[edges.length]; // global visisted

    for(int i=0; i<edges.length; i++){
        if(vis[i]) continue;
        HashMap<Integer, Integer> x = new HashMap<>();  // local visited
        for (int idx=i, dist=0; idx!=-1; idx=edges[idx]){
            if(x.containsKey(idx)){
                res = Math.max(res, dist-x.get(idx));
                break;
            }
            if(vis[idx]) break;
            vis[idx] = true;
            x.put(idx, dist++);
        } 
    }
    return res;
}

Logic:
for every node during our local dfs traversal, we can store the distance from our starting node.
if we find a node that we have aleady visited (we use a map x to check that), there exists a cycle
the length of the cycle would be dist-x.get(idx)

now we don't need to check if a cycle exists for any of the nodes that we visited during our traversal.
why? because since there only exists one directed edge from each node, we're sure that we've already visited any cycle from that node (if it exists).
so we can just ignore that node if we ever see it again.

Intuition

When each node was first visited (visitTime)

If we encounter a node that was visited during the current path, a cycle is detected

By subtracting the visit time of that node from the current time, we get the length of the cycle.

Approach

  1. Initialize:
  • visitTime[]: an array that stores the global visit timestamp of each node.
  • Fill all elements as -1 to mark unvisited nodes.
  • Use time to represent the global visit step.
  • Use longest to track the maximum cycle length found.
  1. Iterate through each node:
  • If already visited, skip.
  • Otherwise, start walking the path from the current node.
  • Track the starting timestamp of this traversal (startTime).
  1. Traverse path:
  • Walk until the node becomes -1 (no outgoing edge), or
  • The node was visited earlier.
  1. Cycle detection:
  • If a visited node is encountered and its visit time is within the current walk (visitTime[node] >= startTime), a cycle is found.
  • Calculate cycle length: time - visitTime[node].
  • Update longest if this cycle is longer.
  1. Return the longest cycle found (or -1 if none found).

Complexity

  • Time complexity: O(n),
    Every node is visited at most once.
    No recursive calls or repeated traversals.
    Total operations are linear in number of nodes.

  • Space complexity: O(n),
    We use: visitTime[] of size n to store the timestamp of visits.

public class Solution {
    public int longestCycle(int[] edges) {
        int n = edges.length;
        int[] visitTime = new int[n]; // Stores global visit time for each node
        Arrays.fill(visitTime, -1);

        int time = 0;
        int longest = -1;

        for (int i = 0; i < n; i++) {
            if (visitTime[i] != -1) continue; // Already visited globally

            int startTime = time;
            int node = i;

            // Walk the path and record visit time
            while (node != -1 && visitTime[node] == -1) {
                visitTime[node] = time++;
                node = edges[node];
            }

            // If we reached a node seen in current path, a cycle is detected
            if (node != -1 && visitTime[node] >= startTime) {
                longest = Math.max(longest, time - visitTime[node]);
            }
        }

        return longest;
    }
}
