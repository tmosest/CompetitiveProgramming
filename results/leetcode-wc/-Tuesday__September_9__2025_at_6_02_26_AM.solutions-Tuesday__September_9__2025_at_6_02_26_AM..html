class Solution {
     public int maxProfit(int[] prices, int[] profits) {
         int len = prices.length;
         int max = -1;
         for(int mid=1; mid<len-1; mid++){

             int p = prices[mid];
              int leftMax = -1;
             for(int left=0; left<mid; left++){
                if(prices[left]<p)
                     leftMax = Math.max(leftMax, profits[left]);
             }
             if(leftMax==-1) continue;


             int rightMax = -1;
             for(int right = mid+1; right<len; right++){
                 if(prices[right]>p){
                     rightMax = Math.max(rightMax, profits[right]);
                 }
             }

            if(rightMax==-1) continue;

             max = Math.max(max, leftMax + rightMax + profits[mid]);
         }
         return max;
     }
}

class Solution {
 public int maxProfit(int[] prices, int[] profits) {

        TreeMap<Integer,Integer> one = new TreeMap<>();
        TreeMap<Integer,Integer> two = new TreeMap<>();

        int n = prices.length;

        int res = -1;

        for(int i=0; i<n; i++){
            var en = two.lowerEntry(prices[i]);
            if(en!=null){
                res = Math.max(res, en.getValue()+profits[i]);
            }

            en = one.lowerEntry(prices[i]);
            if(en != null){
                insert(prices[i],  en.getValue() + profits[i], two);
            }
            insert(prices[i], profits[i], one);
        }

        return res;
    }

    private void insert(int price, int profit, TreeMap<Integer,Integer> map){
        var en = map.floorEntry(price);
        if(en!=null && en.getValue()>=profit) return;
        
        // delete all prices which  is higher than or equal to currPrice and profit is lower than currProfit
        for(Integer p = map.ceilingKey(price); p!=null && map.get(p)<=profit; p = map.higherKey(p)){
            map.remove(p);
        }
        map.put(price, profit);
    }
}

class Solution {
    public int maxProfit(int[] prices, int[] profits) {

        int maxPrice = 0;
        int n = prices.length;
        for(int p : prices){
            maxPrice = Math.max(maxPrice, p); // 5
        }
        FenwickTree fwtLeft = new FenwickTree(maxPrice);
        FenwickTree fwtRight = new FenwickTree(maxPrice);
        
        int[] maxLeft = new int[n]; // ?????leftMax
        for(int i=0; i<n; i++){
            fwtLeft.update(prices[i], profits[i]);
            maxLeft[i] = fwtLeft.get(prices[i]-1);
        }
        // maxLeft??? the max profit of [0, price[i]-1]
        int res = -1;
        for(int i=n-1; i>=0; i--){
            int comp = maxPrice+1 - prices[i];  //find suffixMax by complement, plus 1 to avoid 0
            int maxRight = fwtRight.get(comp-1); // ?<=6?preMax ?????>4?sufMax
            // nums : 10, 2, 3, 4
            // fwtR :  0, 8, 7, 6
            // ex: i=2, ????<=7?preMax => ?????>=4? sufMax
            if(maxLeft[i]>0 && maxRight>0){ 
                res = Math.max(res, maxLeft[i] + profits[i] + maxRight);
            }
            fwtRight.update(comp, profits[i]);
        }
        return res;
    }
}

class FenwickTree {
    public int[] arr;

    public FenwickTree(int n) {
        arr = new int[n+1];
    }

    public FenwickTree(int[] nums) {
        int len = nums.length;
        arr = new int[len + 1];

        for(int i=0; i<len; i++){
            update(i+1, nums[i]);
        }
    }

    // ???? + lowbit
    public void update(int i, int delta) { // Time: O(logN)
        while (i < arr.length) {
            arr[i] = Math.max(arr[i], delta);
            i += lowbit(i);
        }
    }

    public int range(int i, int j) {  // Time: O(logN)
        return get(j)-get(i-1);
    }

    // ???? - lowbit
    public int get(int i) {  // Time: O(logN)
        int sum = 0;
        while (i > 0) {
            sum = Math.max(sum ,arr[i]);
            i -= lowbit(i);
        }
        return sum;
    }

    private static int lowbit(int i) {
        return i & -i;
    }
}

Intuition

For each price we need to find maxProfit in left and right, combine them and then evaluate result with the best. To make sure the current price only evaluates max profits with smaller prices or max profits with larger prices use a bit.

Approach

Create 2 bits, one for max profits to the left and the other for max profits to the right. These bits will cache all possible prices and then using the first bit we process each price[i] and check for max profits seen for price[i] - 1 prices between 0 and i - 1 (inclusive). We do the same thing for the other bit but in reverse so we get the max to the right for prices[i] + 1 prices between i + 1 and n - 1 (inclusive). After this we combine the results and keep track of the max profit.

Complexity

  • Time complexity:
    O(n lg maxPrice)

  • Space complexity:
    O(maxPrice)

Code

//T:O(n lg maxPrice)
//S:O(maxPrice)
class Solution {
    public int maxProfit(int[] prices, int[] profits) {
        int maxPrice = 0, n = prices.length;
        for (int price : prices) {
            maxPrice = Math.max(maxPrice, price);
        }

        int[] bitForMaxLeft = new int[maxPrice + 1], bitForMaxRight = new int[maxPrice + 1];
        int[] maxLeft = new int[n];

        for (int i = 0; i < n; i++) {
            maxLeft[i] = get(bitForMaxLeft, prices[i] - 1);
            update(bitForMaxLeft, prices[i], profits[i]);
        }

        int res = -1;
        for (int i = n - 1; i >= 0; i--) {
            int offset = maxPrice - prices[i] + 1;
            int maxRight = get(bitForMaxRight, offset - 1);

            if (maxLeft[i] > 0 && maxRight > 0) {
                res = Math.max(res, maxLeft[i] + profits[i] + maxRight);
            }
            update(bitForMaxRight, offset, profits[i]);
        }

        return res;
    }

    private void update(int[] bit, int price, int profit) {
        for (int i = price; i > 0 && i < bit.length; i = i + (i & -i)) {
            bit[i] = Math.max(bit[i], profit);
        }
    }

    private int get(int[] bit, int price) {
        int profit = 0;
        for (int i = price; i > 0; i = i - (i & -i)) {
            profit = Math.max(bit[i], profit);
        }

        return profit;
    }
}

Segment Tree solution O(n*log(n))

Segment Tree

Intuition

The task boils down to finding maximum profit for each prices[j]>prices[i] where j>i and maximum profit for each prices[k]<prices[i] where k<i.
So we need to do range queries and updates for each i. Segment tree allows us to do it in O(n*log(n)) time.

Approach

We're going to use 2 Segment trees, each one will store maximum profit in range.
First, we're traversing prices from left to right (this way all the values in the tree are for the indices less than the current one). Saving the maximum profit for each index.
Second, we're traversing prices from right to left (this way all the values in the tree #2 are for the indices higher than the current one). Saving the maximum profit for each index.
Then we're just checking the maximum for each index in one more loop.

Complexity

  • Time complexity:
    O(n*log(n))

Code

class Solution {
    public int maxProfit(int[] prices, int[] profits) {
        int[] arr = new int[prices.length];
        Node root1 = new Node(-1, 0, 1_000_000), root2 = new Node(-1, 0, 1_000_000);
        for (int i = 0; i < prices.length; i++) {
            int q1 = root1.query(0, prices[i] - 1);
            root1.update(prices[i], profits[i]);
            if (q1 != -1 && arr[i] != -1) {
                arr[i] += q1;
            } else if (q1 == -1) {
                arr[i] = -1; 
            }
            int j = prices.length - 1 - i;
            int q2 = root2.query(prices[j] + 1, 1_000_000);
            root2.update(prices[j], profits[j]);
            if (q2 != -1 && arr[j] != -1) {
                arr[j] += q2;
            } else if (q2 == -1) {
                arr[j] = -1; 
            }
        }
        int res = -1;
        for (int i = 0; i < arr.length; i++) {
            if (arr[i] != -1) {
                int tmp = arr[i] + profits[i];
                if (tmp > res) {
                    res = tmp;
                }
            }
        }
        return res;
    }
}

class Node {
    int val;
    int lo;
    int hi;
    int mid;
    Node l, r;
    
    Node(int v, int l, int r) {
        val = v;
        lo = l;
        hi = r;
        mid = lo + (hi - lo) / 2;
    }
    
    void update(int arrIx, int val) {
        if (lo == hi) {
            this.val = Math.max(val, this.val);
            return;
        }
        if (arrIx > mid) {
            if (r == null) {
                r = new Node(-1, mid + 1, hi);
            }
            r.update(arrIx, val);
        } else {
            if (l == null) {
                l = new Node(-1, lo, mid);
            }
            l.update(arrIx, val);
        }
        if (l == null) {
            this.val = r.val;
            return;
        }
        if (r == null) {
            this.val = l.val;
            return;
        }
        this.val = Math.max(l.val, r.val);
    }
    
    int query(long i, long j) {
        if (lo > j || hi < i) {
            return 0;
        }

        if (i <= lo && j >= hi) {
            return val;
        }

        if (i > mid) {
            if (r == null) {
                return -1;
            }
            return r.query(i, j);
        } else if (j <= mid) {
            if (l == null) {
                return -1;
            }
            return l.query(i, j);
        }
        if (r == null && l == null) {
            return val;
        }
        if (r == null) {
            return l.query(i, j);
        }
        if (l == null) {
            return r.query(i, j);
        }
        return Math.max(r.query(i, j), l.query(i, j));
    }
}
