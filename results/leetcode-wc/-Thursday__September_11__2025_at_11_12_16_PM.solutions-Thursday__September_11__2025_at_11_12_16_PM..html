The value of a node is the product of:

  1. number of nodes in the left subtree,
  2. number of nodes in the right subtree,
  3. number of all other nodes, excluding the current one (n - left - right - 1)

We can just use DFS to count child nodes for (1) and (2), and we can then compute (3) as we know the total nubers of nodes.

C++

int dfs(vector<vector<int>> &al, vector<long long> &s, int i) {
    long long prod = 1, sum = 1;
    for (int j : al[i]) {
        int cnt = dfs(al, s, j);
        prod *= cnt;
        sum += cnt;
    }
    s[i] = prod * (max(1ll, (long long)al.size() - sum));
    return i != 0 ? sum : count(begin(s), end(s), *max_element(begin(s), end(s)));
}
int countHighestScoreNodes(vector<int>& p) {
    vector<vector<int>> al(p.size());
    vector<long long> s(p.size());
    for (int i = 1; i < p.size(); ++i)
        al[p[i]].push_back(i);
    return dfs(al, s, 0);
}

Java

long dfs(List<List<Integer>> al, long[] s, int i) {
    long prod = 1, sum = 1;
    for (int j : al.get(i)) {
        long cnt = dfs(al, s, j);
        prod *= cnt;
        sum += cnt;
    }
    s[i] = prod * (Math.max(1, al.size() - sum));
    return sum;
}    
public int countHighestScoreNodes(int[] p) {
    List<List<Integer>> al = new ArrayList<>();
    for (int i = 0; i < p.length; ++i)
        al.add(new ArrayList<Integer>());         
    long[] s = new long[p.length];
    for (int i = 1; i < p.length; ++i)
        al.get(p[i]).add(i);
    dfs(al, s, 0);
    long max_val = Arrays.stream(s).max().getAsLong();
    return (int)Arrays.stream(s).filter(v -> v == max_val).count();
}

class Solution {
    class TreeNode{
        int val;
        TreeNode left;
        TreeNode right;
        TreeNode(int val){
            this.val=val;
            this.left=left;
            this.right=right;
        }
    }
    public int size(TreeNode root){
        if(root==null) return 0;
        else return 1+size(root.left)+size(root.right);
    }
    Map<Long,Integer> mp1=new HashMap<>();
    public void f(TreeNode root,int n){
        if(root==null) return;
        if(root.left==null && root.right==null){
            mp1.put((long)n-1,mp1.getOrDefault((long)n-1,0)+1);
            return;
        }
        int s1=size(root.left);
        int s2=size(root.right);
        int rem=n-s1-s2-1;
        long pro=1;
        if(s1!=0) pro*=s1;
        if(s2!=0) pro*=s2;
        if(rem!=0) pro*=rem;
        mp1.put(pro,mp1.getOrDefault(pro,0)+1);
        f(root.left,n);
        f(root.right,n);
    }
    public int countHighestScoreNodes(int[] par) {
        int n=par.length;
        TreeNode root=new TreeNode(0);
        Map<Integer,List<Integer>> mp=new HashMap<>();
        for(int i=1;i<n;i++){
            if(!mp.containsKey(par[i])){
                List<Integer> l=new ArrayList<>();
                l.add(i);
                mp.put(par[i],l);
            }
            else{
                List<Integer> l=mp.get(par[i]);
                l.add(i);
                mp.put(par[i],l);
            }
        }
        TreeNode newRoot=root;
        Queue<TreeNode> q=new LinkedList<>();
        q.add(root);
        while(!q.isEmpty()){
            TreeNode x=q.poll();
            if(mp.containsKey(x.val)){
                List<Integer> l=mp.get(x.val);
                if(l.size()==1){
                    x.left=new TreeNode(l.get(0));
                    q.add(x.left);
                }
                else if(l.size()==2){
                    x.left=new TreeNode(l.get(0));
                    x.right=new TreeNode(l.get(1));
                    q.add(x.left);
                    q.add(x.right);
                }
            }            
        }
        long pro=1;
        int l=size(root.left);
        int r=size(root.right);
        if(l!=0) pro*=l;
        if(r!=0) pro*=r;
        mp1.put(pro,mp1.getOrDefault(pro,0)+1);
        f(newRoot.left,n);
        f(newRoot.right,n);
        long max=0;
        int ans=0;
        for(var e:mp1.keySet()){
            if(max<e){
                max=Math.max(max,e);
                ans=mp1.get(e);
            }
        }
        return ans;
    }
}

Optimised Code

class Solution {
    private int[] count;
    private int n;
    private long maxScore = 0;
    private int result = 0;

    public int countHighestScoreNodes(int[] parents) {
        n = parents.length;
        List<Integer>[] children = new ArrayList[n];
        for (int i = 0; i < n; i++) {
            children[i] = new ArrayList<>();
        }

        for (int i = 1; i < n; i++) {
            children[parents[i]].add(i);
        }

        count = new int[n];
        dfs(0, children);

        return result;
    }

    private int dfs(int node, List<Integer>[] children) {
        long score = 1;
        int total = 0;

        for (int child : children[node]) {
            int sz = dfs(child, children);
            score *= sz;
            total += sz;
        }

        int rest = n - total - 1;
        if (rest > 0) score *= rest;

        if (score == maxScore) {
            result++;
        } else if (score > maxScore) {
            maxScore = score;
            result = 1;
        }

        count[node] = total + 1;
        return count[node];
    }
}

class Solution {
    long max = 0, res = 0;
    public int countHighestScoreNodes(int[] parents) {
        Map<Integer, List<Integer>> hm = new HashMap();
        for(int i = 0; i < parents.length; i++) {   // build the tree
            hm.computeIfAbsent(parents[i], x ->new ArrayList<>()).add(i);
        }
        dfs(0, parents.length, hm); // traverse the tree to get the result
        return (int)res;
    }
    int dfs(int s, int n, Map<Integer, List<Integer>> hm) {
        int sum = 1;
        long mult = 1L;
        for(int child : hm.getOrDefault(s, new ArrayList<>())) {
            int count = dfs(child, n, hm); // subtree node count
            sum += count;
            mult *= count; // multiply the result by children size
        }     
        mult *= (s == 0 ? 1L : n - sum); // multiply the result by remain size except self and children size(the nodes through parent)
        if(mult > max) {
            max = mult;
            res = 1;
        } else if (mult == max) {
            res++;
        }
        return sum; // return the node count of the tree rooted at s
    }
}
