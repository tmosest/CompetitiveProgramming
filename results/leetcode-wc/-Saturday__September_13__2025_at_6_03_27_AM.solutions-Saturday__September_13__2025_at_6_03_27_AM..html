class Solution {
    const int M = 1000000007, RM = (M + 1) / 2;
    
    int mul(long long x, long long y) {
        return x * y % M;
    }
    
    int sqr(int x) {
        return mul(x, x);
    }
    
    int add(int x, int y) {
        if ((x += y) >= M) {
            x -= M;
        }
        return x;
    } 
    
    int sub(int x, int y) {
        if ((x -= y) < 0) {
            x += M;
        }
        return x;
    }
    
    int C2(int n) {
        return mul(mul(n, n - 1), RM);
    }
    
    int count(const vector<int> &v, const vector<int> &c, bool countIfEqual) {
        const int n = v.size(), m = c.size();
        vector<int> t(m);
        int r = 0, R2 = 0;
        for (int i = 0; i < m; ++i) {
            R2 = add(R2, sqr(c[i]));
        }
        int RT = 0, R2T = 0;
        for (int i = 0; i < n; ++i) {
            const int x = v[i];
            int rx = c[x] - t[x];
            //left = 1, right = 0
            const int r2 = sub(R2, sqr(rx)), rt = sub(RT, mul(rx, t[x])), r2t = sub(R2T, mul(sqr(rx), t[x])), p = n - i - rx, sumt = i - t[x];
            // Sigma((right_no_x - right[j]) ^ 2) - (r2_no_x - right[j] ^ 2) * t[j] * t[x] / 2
            // t[x] / 2 can be done finally
            // P = n - i - right[x]
            //((P - right[j]) ^ 2 - right2 + right[x] ^ 2 + right[j] ^ 2) * t[j]
            // (P ^ 2 - 2 * P * right[j] + 2 * right[j] ^ 2 +  - r2_no_x) * t[j]
            int temp = mul(sub(sqr(p), r2), sumt);
            temp = sub(temp, mul(mul(2, p), rt));
            temp = add(temp, mul(2, r2t));
            temp = mul(temp, mul(t[x], RM));
            r = add(r, temp);
            //left = 2, right = 0
            r = add(r, mul(C2(t[x]), C2(p)));
            //left = 2, right = 1
            --rx;
            r = add(r, mul(C2(t[x]), mul(rx, p)));
            if (countIfEqual) {
                // left = right = 1
                r = add(r, mul(mul(t[x], sumt), mul(rx, p)));
                // left = right = 2
                r = add(r, mul(C2(t[x]), C2(rx)));
            }  
            ++t[x];
            R2 = add(r2, sqr(rx));
            RT = add(rt, mul(rx, t[x]));
            R2T = add(r2t, mul(sqr(rx), t[x]));
        }
        return r;
        
    }
    
public:
    int subsequencesWithMiddleMode(vector<int>& nums) {
        unordered_map<int, int> mp;
        int m = 0;
        for (int& x : nums) {
            if (!mp.count(x)) {
                mp[x] = m++;
            }
            x = mp[x];
        }
        vector<int> c(m);
        for (int x : nums) {
            ++c[x];
        }
        int r = count(nums, c, true);
        reverse(nums.begin(), nums.end());
        return add(r, count(nums, c, false));
    }
};

Note: I wrote a solution for the previous version of this problem (Problem 3395). The solution below is identical to that one, except that it uses the inbuilt Python function comb, instead of the custom one I wrote for that problem. This, plus using Python 3, makes a difference between Accepted and Time Limit Exceeded for the present problem.

Intuition

First, I'd like to give credit to user Ye Gao, who wrote a post with a very similar solution. I think the value of my solution lies in clear variable names and comments, which make it longer but hopefull also easier to follow the logic of the code.

The basic idea of the solution is to count the number of subsequences with a given middle index i that satisfy the conditions of the problem. For each index, this involves careful counting, and the use of the inclusion-exclusion principle. Along the way, we keep track of the count of occurrences of each number to the left and right of i, as well as the number of pairs of unequal numbers on both sides of i.

I don't think this problem is difficult as much as it is annoying, in that it requires a lot of care to make sure that the computation is done carefully, without missing any cases and without double-counting anything.

Approach

For each value of i, we denote num = nums[i] and do the following:

  • Compute the number of subsequences with one occurrence of num to the left of i and three other distinct numbers;
  • Compute the number of subsequences with one occurrence of num to the right of i and three other distinct numbers;
  • Compute the number of subsequences with three of more total occurrences of num;
  • Add these results to get the number of subsequences with a unique middle mode at index i, and add to the overall total.

Each of these cases is broken down into a few subcases, that need to be kept track of carefully.

The original submission is here, and the code with detailed comments is below.

Complexity

  • Time complexity: the processing time of each index i will depend on how many different values there are in the original list, which could be O(N) in the worst case, so it could be O(N2) in the worst case. If the number of different values in the original list is smaller than O(N), this will be faster.

  • Space complexity: we need to keep two counters and a few extra variables, so the space complexity is O(N).

class Solution:
    def subsequencesWithMiddleMode(self, nums: List[int]) -> int:
        p = 10 ** 9 + 7
        n = len(nums)

        # Variable meanings:
        # > leftCounter, rightCounter are Counters for numbers to the
        #       left and right of current index
        # > leftPairs, rightPairs store the number of non-equal pairs
        #       of numbers to the left and right of current index
        # > numSubsequences will store the final result
        leftCounter, rightCounter = Counter(), Counter(nums)
        leftPairs, rightPairs = 0, comb(n, 2) - sum([comb(val, 2) for val in rightCounter.values()])
        numSubsequences = 0

        for i, num in enumerate(nums): 
            # In this iteration, will consider subsequences  
            # whose unique mode is nums[i], or num

            # *** Part 1: bookkeeping rightCounter, rightPairs ***
            rightPairs -= (n - i - rightCounter[num])
            rightCounter[num] -= 1

            # The variable auxCount stores the number of subsequences
            # in which i is the middle index and num is a unique mode
            auxCount = 0 

            # *** Part 2: computing number of subsequences with exactly two instances of ***
            # ***         num, and no more than one instance of other numbers            ***

            # If num appears to the left of i:
            if leftCounter[num]: 
                for num2 in leftCounter:
                    # Consider values that appear to the left of i which are not equal to num
                    if num != num2 and leftCounter[num2] > 0:
                        # Start with the number of pairs of unequal numbers to the right of i
                        auxVal = rightPairs

                        # Subtract count of pairs to the right of i in which num appears
                        auxVal -= rightCounter[num] * (n - 1 - i - rightCounter[num])

                        # Subtract count of pairs to the right of i in which num2 appears
                        auxVal -= rightCounter[num2] * (n - 1 - i - rightCounter[num2])

                        # Add back count of pairs to the right of i in which both num and num2 appear
                        auxVal += rightCounter[num] * rightCounter[num2]

                        # Multiply by number of pairs to the left of i in which both num and num2 appear,
                        # and add to the total count of subsequences                    
                        auxCount += auxVal * leftCounter[num] * leftCounter[num2]

            # If num appears to the right of i:
            if rightCounter[num]: 
                for num2 in rightCounter:
                    # Consider values that appear to the right of i which are not equal to num 
                    if num != num2 and rightCounter[num2] > 0:
                        # Start with the number of pairs of unequal numbers to the left of i
                        auxVal = leftPairs

                        # Subtract count of pairs to the left of i in which num appears
                        auxVal -= leftCounter[num] * (i - leftCounter[num])
                        
                        # Subtract count of pairs to the left of i in which num2 appears 
                        auxVal -= leftCounter[num2] * (i - leftCounter[num2])
                        
                        # Add back count of pairs to the left of i in which both num and num2 appear
                        auxVal += leftCounter[num] * leftCounter[num2]
                        
                        # Multiply by number of pairs to the right of i in which both num and num2 appear,
                        # and add to the total count of subsequences
                        auxCount += auxVal * rightCounter[num] * rightCounter[num2]

            # *** Part 3: Computing number of subsequences with three or more occurrence of num ***
            # Sequences with two occurrences of num to the left of i and any number of occurrences of num to the right of i
            auxCount += comb(leftCounter[num], 2) * comb(n - 1 - i, 2)

            # Sequences with one occurrence of num to the left of i and at least one occurrence of num to the right of i
            auxCount += leftCounter[num] * (i - leftCounter[num]) * (comb(n - i - 1, 2) - comb(n - i - 1 - rightCounter[num], 2))

            # Sequences with no occurrences of num to the left of i and two occurrences of num to the right of i
            auxCount += comb(i - leftCounter[num], 2) * comb(rightCounter[num], 2)
            
            # *** Part 3: Add current count to overall result ***
            numSubsequences += auxCount
            
            # *** Part 4: bookkeeping leftCounter, leftPairs ***
            leftPairs += i - leftCounter[num]
            leftCounter[num] += 1
        
        return numSubsequences % p

from collections import Counter
from math import comb

class Solution:

    def __init__(self):
        self.total = 0

        # Pre computed sums...
        self.sum_prefix_squared = 0
        self.sum_suffix_squared = 0
        self.sum_prefix_and_suffix = 0
        self.sum_prefix_squared_and_suffix = 0
        self.sum_suffix_squared_and_prefix = 0

        # Counters
        self.prefix_to_counts = {}
        self.suffix_to_counts = {}

    # For Debug...
    # def printSums(self, indent:int = 0) -> None:
    #     indent_helper = '\t' * indent
    #     print(f'{indent_helper}the sums are...')
    #     print(f'{indent_helper}\tsum prefix^2...........:{self.sum_prefix_squared}')
    #     print(f'{indent_helper}\tsum suffix^2...........:{self.sum_prefix_squared}')
    #     print(f'{indent_helper}\tsum prefix and suffix..:{self.sum_prefix_squared}')
    #     print(f'{indent_helper}\tsum prefix^2 and suffix:{self.sum_prefix_squared}')
    #     print(f'{indent_helper}\tsum suffix^2 and prefix:{self.sum_prefix_squared}')

    def incrementSums(self, key_to_counts: int) -> None:
        """Increases prefix and suffix sum of counts with key_to_counts as key."""
        counts_prefix = self.prefix_to_counts[key_to_counts]
        counts_prefix_squared = counts_prefix * counts_prefix
        counts_suffix = self.suffix_to_counts[key_to_counts]
        counts_suffix_squared = counts_suffix * counts_suffix

        self.sum_prefix_squared += counts_prefix_squared
        self.sum_suffix_squared += counts_suffix_squared
        self.sum_prefix_and_suffix += counts_prefix * counts_suffix
        self.sum_prefix_squared_and_suffix += counts_prefix_squared * counts_suffix
        self.sum_suffix_squared_and_prefix += counts_suffix_squared * counts_prefix

    def decrementSums(self, key_to_counts: int) -> None:
        """Decreases prefix and suffix sum of counts with key_to_counts as key."""
        counts_prefix = self.prefix_to_counts[key_to_counts]
        counts_prefix_squared = counts_prefix * counts_prefix
        counts_suffix = self.suffix_to_counts[key_to_counts]
        counts_suffix_squared = counts_suffix * counts_suffix

        self.sum_prefix_squared -= counts_prefix_squared
        self.sum_suffix_squared -= counts_suffix_squared
        self.sum_prefix_and_suffix -= (counts_prefix * counts_suffix)
        self.sum_prefix_squared_and_suffix -= (counts_prefix_squared * counts_suffix)
        self.sum_suffix_squared_and_prefix -= (counts_suffix_squared * counts_prefix)

    def subsequencesWithMiddleMode(self, nums: List[int]) -> int:
        max_mod = 1_000_000_007
        n = len(nums)

        @cache
        def chooseTwo(val: int) -> int:
            return math.comb(val, 2)

        def addToTotal(val: int) -> int:
            """Adds to total with modulus applied after."""
            # print(f'ADDING {val}...')
            self.total += val
            self.total %= max_mod
            # print(f'...........CURRENT_TOTAL: {self.total}')
            return self.total

        # For Debug...
        # def checkCounts(indent: int = 0) -> None:
        #     indent_helper = '\t' * indent
        #     print(f'{indent_helper}prefix_to_counts...')
        #     for each_key, each_count in self.prefix_to_counts.items():
        #         print(f'{indent_helper}\t{each_key} prefix occurs {each_count} times')
        #     print(f'{indent_helper}suffix_to_counts...')
        #     for each_key, each_count in self.suffix_to_counts.items():
        #         print(f'{indent_helper}\t{each_key} suffix occurs {each_count} times')

        self.prefix_to_counts = Counter()
        self.suffix_to_counts = Counter(nums)


        # Initialize the suffix squared total:
        for each_value in self.suffix_to_counts.values():
            squared = math.pow(each_value, 2)
            self.sum_suffix_squared += squared
    
        # checkCounts()
        for i, each_val in enumerate(nums):
            # Update prefix and suffix sums
            # print(f'Looking at nums[i]={i}: {each_val}...')
            # self.printSums()
            # Remove current suffix...
            self.decrementSums(each_val)
            # Update suffix counts...
            self.suffix_to_counts[each_val] -= 1
            # print(f'\tdecrement {each_val} suffix counts to {self.suffix_to_counts[each_val]}.')
            # Update sums...
            self.incrementSums(each_val)


            current_prefix_counts = self.prefix_to_counts[each_val]
            current_suffix_counts = self.suffix_to_counts[each_val]

            # checkCounts()
            # print(f'\t\tCurrent prefix={each_val} occurs {current_prefix_counts} times to the left')
            # print(f'\t\tCurrent suffix={each_val} occurs {current_suffix_counts} times to the right')

            left = i
            right = n - i - 1

            # print(f'\t\t\tThere are {left} slots to the left, and {right} slots to the right...')
            all_the_ways = chooseTwo(left) * chooseTwo(right)
            # print(f'\t\t\t...which means all the ways to count total {all_the_ways}...')

            addToTotal(all_the_ways)

            left_overcount = left - current_prefix_counts
            # print(f'\t\t\t\twe have to remove/adjust for the extra {left_overcount} slots to the left...')
            right_overcount = right - current_suffix_counts
            # print(f'\t\t\t\twe have to remove/adjust for the extra {right_overcount} slots to the right...')
            self.total -= chooseTwo(left_overcount) * chooseTwo(right_overcount)

            # Adjust total for edge cases...
            self.decrementSums(each_val)

            # Edge cases...
            #     ...b   b  a  a  *R
            #     ...a  *L  a  b   b
            #     ...a   b  a  b  *R
            #     ...b  *L  a  a   b
            self.total -= int(
                (self.sum_prefix_squared - left_overcount) * current_suffix_counts * right_overcount // 2
                )
            self.total -= int(
                (self.sum_suffix_squared - right_overcount) * current_prefix_counts * left_overcount // 2
                )
            
            self.total -= self.sum_prefix_and_suffix * (
                (current_prefix_counts  * right)
                - (2 * current_prefix_counts * current_suffix_counts)
                + (current_suffix_counts * left)
                )

            # Edge cases...
            #     ...bb aa b
            #     ...a b a bb
            addToTotal(self.sum_prefix_squared_and_suffix * current_suffix_counts)
            addToTotal(self.sum_suffix_squared_and_prefix * current_prefix_counts)
    
            # Add to prefix and adjust sums...
            self.prefix_to_counts[each_val] += 1
            # print(f'increment {each_val} counts to {self.prefix_to_counts[each_val]}.')
            self.incrementSums(each_val)

        return self.total
