class Solution:
    def maximumSubarraySum(self, nums: List[int], k: int) -> int:
        n = len(nums)
        elements = set()
        current_sum = 0
        max_sum = 0
        begin = 0

        for end in range(n): 
            if nums[end] not in elements:
                current_sum += nums[end]
                elements.add(nums[end])

                if end - begin + 1 == k:
                    if current_sum > max_sum:
                        max_sum = current_sum
                    
                    current_sum -= nums[begin]
                    elements.remove(nums[begin])
                    begin += 1
            else:
                while nums[begin] != nums[end]:
                    current_sum -= nums[begin]
                    elements.remove(nums[begin])
                    begin += 1
                
                begin += 1

        return max_sum

class Solution:
    def maximumSubarraySum(self, nums: List[int], k: int) -> int:
        res = 0
        count = defaultdict(int)
        cur_sum = 0

        l = 0
        for r in range(len(nums)):
            cur_sum += nums[r]
            count[nums[r]] += 1

            if r - l + 1 > k:
                count[nums[l]] -= 1
                if count[nums[l]] == 0:
                    count.pop(nums[l])
                cur_sum -= nums[l]
                l += 1

            if len(count) == r - l + 1 == k:
                res = max(res, cur_sum)

        return res

class Solution:
    def maximumSubarraySum(self, nums: List[int], k: int) -> int:
        res = 0
        prev_idx = {}  # num -> prev index of num
        cur_sum = 0

        l = 0

        for r in range(len(nums)):
            cur_sum += nums[r]

            i = prev_idx.get(nums[r], -1)

            while l <= i or r - l + 1 > k:
                cur_sum -= nums[l]
                l += 1

            if r - l + 1 == k:
                res = max(res, cur_sum)

            prev_idx[nums[r]] = r

        return res

Aproach #1

  • We need to find a sliding window of size k where all elements are distinct
  • For each valid window, we trac the sum and keep updating the maximum sum found
  • When we encounter duplicates, we need to adjust our window by removing elements until we eliminate the duplicate

Execution

  1. Use a sliding window technique with two pointers: 'begin' and 'end'
  2. Maintain a set to track unique elements in current window
  3. Keep track of current sum and maximum sum found so far
  4. For each element at 'end' pointer:
    • If element is unique:
      • Add it to sum and set
      • If window size reaches k, update max_sum if needed and slide window
    • If element is duplicate:
      • Remove elements from start until duplicate is removed
      • Adjust window start accordingly

Sliding Window
C++
Java

? Intuition

We want exactly k distinct elements in a window.

A brute force solution would check all n-k+1 subarrays, but that’s O(n·k) ? too slow.

Instead, use a sliding window with a HashSet:

Expand the window with r.

If a duplicate appears, shrink from the left (l) until valid.

When the window has exactly k distinct elements ? update the max sum.

After checking, shrink from left to continue exploring further subarrays.

This way, each element is added and removed at most once, giving us an O(n) solution.

?? Algorithm

Initialize:

Two pointers: l = 0, r = 0 (window boundaries).

A HashSet to store unique elements in the current window.

Variables sum = 0, maxSum = 0.

Iterate with r from 0 to nums.length - 1:

Add nums[r] to sum.

If nums[r] is already in the set ? shrink window by removing from l until nums[r] becomes unique.

Insert nums[r] into the set.

If the set size equals k:

Update maxSum.

Then shrink window from l (so future windows can be checked).

Return maxSum.

? Complexity

Time: O(n) ? each element is processed at most twice (once added, once removed).

Space: O(k) for the HashSet.

import java.util.*;

class Solution {
    public long maximumSubarraySum(int[] nums, int k) {
        int l = 0, r = 0;
        Set<Integer> set = new HashSet<>();
        long maxSum = 0, sum = 0;

        while (r < nums.length) {
            sum += nums[r];

            // shrink until nums[r] is unique
            while (set.contains(nums[r])) {
                sum -= nums[l];
                set.remove(nums[l]);
                l++;
            }

            set.add(nums[r]);

            // if window has exactly k distinct numbers
            if (set.size() == k) {
                maxSum = Math.max(maxSum, sum);

                // shrink from left to keep window moving
                sum -= nums[l];
                set.remove(nums[l]);
                l++;
            }

            r++;
        }

        return maxSum;
    }
}
