Approach

The idea behind the solution is pretty straightforward: place all ranges on the interval from 0 to n - 1 and find uncovered sub-ranges by comparing lefts and rights of adjacent sub-ranges.

The best way to order ranges is to sort them by left or right limit in ascending order:

As you can see in the picture, the sorting makes uncovered cells pretty visible. To collect the gaps between sub-ranges we are going to introduce a pointer that will run from the first to last element.

To move the pointer we are going to iterate over the sorted gaps and compare a distance between the pointer and the left limit of the sub-range. If the pointer is located on the left of the range we have a new uncovered cells:

If the pointer is located on the right of the left range limit, it means there is no uncovered cells.

As a last step we move the pointer to a new position: next to the right limit of the current sub-range:

After all sub-ranges passed could have some uncovered elements between the pointer and the last element, so we should add them to the result as well.

Complexity

  • Time complexity: O(nlogn)

  • Space complexity: O(n)

class Solution {
    fun findMaximalUncoveredRanges(n: Int, ranges: Array<IntArray>) = 
        mutableListOf<IntArray>().apply {
            ranges.sortBy { it[0] }
            var ptr = 0
            for ((left, right) in ranges) {
                if (ptr < left) {
                    add(intArrayOf(ptr, left - 1))
                }
                if (ptr <= right) {
                    ptr = right + 1
                }
            }
            if (ptr < n) {
                add(intArrayOf(ptr, n - 1))
            }
        }.toTypedArray()
}

class Solution {
    public int[][] findMaximalUncoveredRanges(int n, int[][] ranges) {
        List<int[]> result = new ArrayList<>();

        Arrays.sort(ranges, (a, b) -> a[0] - b[0]);
        int ptr = 0;
        for (int[] range : ranges) {
            int left = range[0];
            int right = range[1];
            
            if (ptr < left) {
                result.add(new int[] { ptr, left - 1 });
            }
            if (ptr <= right) {
                ptr = right + 1;
            }
        }
        if (ptr < n) {
            result.add(new int[] { ptr, n - 1 });
        }
        return result.toArray(new int[result.size()][]);
    }
}

class Solution {
public:
    vector<vector<int>> findMaximalUncoveredRanges(int n, vector<vector<int>>& ranges) {
        sort(ranges.begin(), ranges.end(), [&](auto& lhs, auto& rhs) {return lhs[1] < rhs[1];}); 
        vector<pair<int, int>> stk; 
        for (auto& r : ranges) {
            int x = r[0], y = r[1]; 
            while (stk.size() && x <= stk.back().second) {
                auto [xx, _] = stk.back(); stk.pop_back(); 
                x = min(x, xx); 
            }
            stk.emplace_back(x, y); 
        }
        vector<vector<int>> ans; 
        int prev = 0; 
        for (auto& [x, y] : stk) {
            if (prev < x) ans.push_back({prev, x-1}); 
            prev = max(prev, y+1); 
        }
        if (prev <= n-1) ans.push_back({prev, n-1}); 
        return ans; 
    }
};

class Solution {
    public int[][] findMaximalUncoveredRanges(int n, int[][] ranges) {
        Arrays.sort(ranges, (x, y)->Integer.compare(x[1], y[1])); 
        Stack<int[]> stk = new Stack(); 
        for (var r : ranges) {
            while (!stk.isEmpty() && r[0] <= stk.peek()[1]) {
                var elem = stk.pop(); 
                r[0] = Math.min(r[0], elem[0]); 
            }
            stk.add(r); 
        }
        List<int[]> ans = new ArrayList(); 
        int prev = 0; 
        for (var elem : stk) {
            if (prev < elem[0]) ans.add(new int[]{prev, elem[0]-1}); 
            prev = Math.max(prev, elem[1]+1); 
        }
        if (prev <= n-1) ans.add(new int[]{prev, n-1}); 
        return ans.toArray(new int[0][0]); 
    }
}

class Solution {
    public int[][] findMaximalUncoveredRanges(int n, int[][] ranges) {
        if(ranges == null || n == 0)
            return new int[0][0];
        if(ranges.length == 0)
            return new int[][] { {0, n-1} };

        Arrays.sort(ranges, (a,b) -> a[0] == b[0] ? b[1] - a[1] : a[0] - b[0]);

        List<int[]> result = new ArrayList<>();
        int lo = 0;
        int hi = n-1;
        int len = ranges.length;

        if(lo < ranges[0][0]) {
            result.add(new int[] { lo, ranges[0][0]-1 } );
            lo = ranges[0][1] + 1;
        }

        for(int i=0; i<len; i++) {
            if(lo < ranges[i][0] && ranges[i][0] <= hi) {
                result.add(new int[] { lo, ranges[i][0]-1 } );
            }
            lo = Math.max(lo, ranges[i][1]+1);
        }

        if(lo <= hi) {
            result.add(new int[] { lo, hi } );
        }

        int[][] res = new int[result.size()][2];
        for(int i=0; i<res.length; i++) {
            res[i] = result.get(i);
        }

        return res;
    }
}
