Sliding Window
Python
C++
Java

Intuition

Use a sliding window of length k to simulate removing each possible substring.

Explanation

Initialize (x, y) = (0, 0)
Track the net change in x and y when a substring is removed.

Initialize seen = {(0,0)}
Maintain a set of visited coordinates to count distinct final positions.

Slide the window over s,
adding the effect of the new move and removing the old move.

Return seen size as the result.

Complexity

Time O(n - k)
Space O(n - k)

Python
Java
C++
    def distinctPoints(self, s: str, k: int) -> int:
        seen = {(0, 0)}
        x = y = 0
        for i in range(k, len(s)):
            if s[i] == 'U': y += 1
            if s[i] == 'D': y -= 1
            if s[i] == 'L': x += 1
            if s[i] == 'R': x -= 1
            if s[i - k] == 'U': y -= 1
            if s[i - k] == 'D': y += 1
            if s[i - k] == 'L': x -= 1
            if s[i - k] == 'R': x += 1
            seen.add((x, y))
        return len(seen)

class Solution {
    public int distinctPoints(String s, int k) {
        if(k==s.length()) {
            return 1;
        }
        Set<String> st = new HashSet<>();
        int secondSequenceOfCordinateX = 0;
        int secondSequenceOfCordinateY = 0;
        for(int i=k;i<s.length();i++) {
           int cordinate[] = calculateCordinate(secondSequenceOfCordinateX, secondSequenceOfCordinateY, s.charAt(i));
           secondSequenceOfCordinateX = cordinate[0];
           secondSequenceOfCordinateY = cordinate[1];
        }
        st.add(secondSequenceOfCordinateX+","+secondSequenceOfCordinateY);
        int l = 0;
        int r = k-1; 
        int firstSequenceCordinateX = 0;
        int firstSequenceCordinateY = 0;
        while(r<s.length()) {
          int cordinates[] = calculateCordinate(firstSequenceCordinateX, firstSequenceCordinateY, s.charAt(l));
          firstSequenceCordinateX = cordinates[0];
          firstSequenceCordinateY = cordinates[1];
          l++;
          r++;
          if(r>=s.length()) {
            break;
          }
          if(s.charAt(r)=='L') {
            secondSequenceOfCordinateX = secondSequenceOfCordinateX + 1;
          }
          if(s.charAt(r)=='R') {
            secondSequenceOfCordinateX = secondSequenceOfCordinateX - 1;
          }
          if(s.charAt(r)=='U') {
            secondSequenceOfCordinateY = secondSequenceOfCordinateY - 1;
          }
          if(s.charAt(r)=='D') {
            secondSequenceOfCordinateY = secondSequenceOfCordinateY + 1;
          }
          //accurate displacement
          int resultX = firstSequenceCordinateX+secondSequenceOfCordinateX;
          int resultY = firstSequenceCordinateY+secondSequenceOfCordinateY;
          String str = resultX+","+resultY;
          st.add(str);
        }
      return st.size();    
    }

   public int[]  calculateCordinate(int x, int y, char ch ) {
      if(ch=='U') {
         y++;
      }
      else if(ch=='D') {
        y--;
      }
      else if(ch=='L') {
         x--;
      }
      else if(ch=='R') {
         x++;
      }
     int cordinate[] = new int[2];
     cordinate[0] = x;
     cordinate[1] = y;
     return cordinate; 
   }
}
