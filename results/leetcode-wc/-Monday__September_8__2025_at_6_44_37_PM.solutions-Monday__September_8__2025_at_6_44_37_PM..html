class Solution {
    public List<Boolean> getResults(int[][] queries) {
        TreeMap<Integer, Integer> obstacles = new TreeMap<>(); // max up until block at key

        List<Boolean> ans = new ArrayList<>();

        for (int[] q : queries) {
            if (q[0] == 1) { // build obstacle
                int index = q[1];
                Integer prev = obstacles.lowerKey(index);
                Integer next = obstacles.higherKey(index);

                if (prev == null) {
                    obstacles.put(index, index);
                } else {
                    obstacles.put(index, Math.max(index - prev, obstacles.get(prev)));
                }

                prev = index;
                while (next != null && obstacles.get(next) > obstacles.get(prev)) {
                    obstacles.put(next, Math.max(next - prev, obstacles.get(prev)));
                    prev = next;
                    next = obstacles.higherKey(next);
                }

            } else { // ans query
                int index = q[1], sz = q[2];
                if (obstacles.containsKey(index)) { // there's an obstacle at queried index
                    ans.add(obstacles.get(index) >= sz);
                } else { // no obstacle
                    Integer prev = obstacles.lowerKey(index);
                    if (prev == null) {
                        ans.add(index >= sz);
                    } else {
                        ans.add(Math.max(index - prev, obstacles.get(prev)) >= sz);
                    }
                }

            }
        }

        return ans;

    }

}

Intution

  • According to the contraints given in the problem its possible to store all data regarding x coordinates in a array.
  • Lets have a array M such that M[i] = largest size block that can be placed considering i to be the ending position of that box.

Lets look at a dry run of this array M on example 2 queries = [[1,7],[2,7,6],[1,2],[2,7,5],[2,7,6]]
Intial values
M = [0,1,2,3,4,5,6,7,8]

Query 1 (1,7) Add an obstacle at x = 7
M = [0,1,2,3,4,5,6,0,1]

Query 3 (1,2) Add an obstacle at x = 2
M = [0,1,0,1,2,3,4,0,1]

Observe at all coordinates i including x, M[i] -= M[x]. Here i is in range of [x,next obstacle after x - 1]

To answer queries of type 2 lets find the maximum in range [0,x], lets name is max

  • if max >= sz there always exists position before x where sz sized block can be placed. answer true.
  • if max == sz-1 now we can consider the right end of the coordinate where there value would have been max value of M, as placing a box which touches a obstacle is allowed. But there is a catch here, if x is the only coordinate with M[x] == max then the answer is false, else answer is true
  • else answer is false.

This can be implemented with a lazy segment tree (Range Add, Range Maximum). I also used a treeSet for implementation.

Please do refer the code provided below, I you have any doubts please feel free to comment.

class Solution{
    public List<Boolean> getResults(int[][] queries) {
        int N = 0; //Maximum value of x coordinate
        for(int[] q: queries) N = Math.max(N,q[1]);
        List<Boolean> answers = new ArrayList<>();
        TreeSet<Integer> treeSet = new TreeSet<>();
        MassAddMaximumSegmentTree M = new MassAddMaximumSegmentTree(N+2);
        for(int i=0;i<N+2;i++) M.addToRange(i,i,i); //Initialize M array
        for(int[] q: queries){
            int x = q[1];
            if(q[0] == 2){
                int sz = q[2];
                long max = M.getMaximumInRange(0,x);
                if(max >= sz) answers.add(true); //case 1
                else if(max == sz - 1){
                    if(M.getMaximumInRange(x,x) != max) answers.add(true); //There exists a place left to x where M[i] is max
                    else{ //Now M[x] == max, lets check if there exists some i where M[i] == max
                        Integer PREV = treeSet.floor(x);
                        if(PREV != null){
                            if(M.getMaximumInRange(0,PREV) >= sz+1) answers.add(true);
                            else answers.add(false);
                        }
                        else answers.add(false); //There were no obstacles in range (0,x) so answer is false
                    }
                }
                else answers.add(false);
            }
            else{
                Integer NEXT = treeSet.ceiling(x);
                int next;
                if(NEXT == null) next = N+1;
                else next = NEXT;
                M.addToRange(x,next-1,-1*(M.getMaximumInRange(x,x))); //Subtract the contribution of M[x]
                treeSet.add(x);
            }
        }
        return answers;
    }
}
class MassAddMaximumSegmentTree {
    long[] segmentArray;
    long[] lazy;
    int length;

    public MassAddMaximumSegmentTree(int n){
        this.length = n;
        int height = (int)Math.ceil(Math.log(length)/Math.log(2));
        int lengthOfSegmentTree = 2*(1<<height)-1;
        this.segmentArray = new long[lengthOfSegmentTree];
        this.lazy = new long[lengthOfSegmentTree];
    }

    public void addToRange(int left, int right, long value){
        add(0,length-1,0,left,right,value);
    }
    private long add(int ss, int se, int si, int qs, int qe, long value){
        if(se<qs || qe<ss) return segmentArray[si];
        if(ss != se) propagate(si);
        if(qs<=ss && se<=qe){
            lazy[si] += value;
            return segmentArray[si] = segmentArray[si]+value;
        }
        int mid = (ss+se)>>1;
        return segmentArray[si] = Math.max(add(ss,mid,2*si+1,qs,qe,value),add(mid+1,se,2*si+2,qs,qe,value));
    }

    public long getMaximumInRange(int left, int right){
        return getMax(0,length-1,0,left,right);
    }

    private long getMax(int ss, int se, int si, int qs, int qe){
        if(se<qs || qe<ss) return Long.MIN_VALUE;
        if(ss != se) propagate(si);
        if(qs<=ss && se<=qe) return segmentArray[si];
        int mid = (ss+se)>>1;
        return Math.max(getMax(ss,mid,2*si+1,qs,qe), getMax(mid+1,se,2*si+2,qs,qe));
    }


    private void propagate(int si){
        if(lazy[si]!=0){
            segmentArray[2*si+1] += lazy[si];
            segmentArray[2*si+2] += lazy[si];
            lazy[2*si+1] += lazy[si];
            lazy[2*si+2] += lazy[si];
            lazy[si] = 0;
        }
    }
}

Intuition

Step-by-Step Explanation:

  1. Class Definition:

    • Define a static Interval class to represent a range [start, end].

  2. getResults Method:

    • Initialize a TreeSet<Interval> to keep track of available intervals.
    • Initialize the TreeSet with an interval representing the entire number line [0, Integer.MAX_VALUE].
    • Initialize a TreeSet<Integer> to store the positions of obstacles.
    • Iterate through the queries and process each query based on its type.

  3. addObstacle Method:

    • Parameters: obstacles (TreeSet of existing obstacles), intervals (TreeSet of current intervals), obstacle (position of the new obstacle).
    • Find the nearest existing obstacle positions to determine the bounds of the interval containing the new obstacle:
      • Use floor(obstacle) to find the largest existing obstacle less than or equal to obstacle.
      • Use ceiling(obstacle) to find the smallest existing obstacle greater than or equal to obstacle.
    • If no such bounds exist, use 0 for the lower bound and Integer.MAX_VALUE for the upper bound.
    • Remove the interval [lowerBound, upperBound] that contains the new obstacle.
    • Add two new intervals: [lowerBound, obstacle] and [obstacle, upperBound].
    • Add the new obstacle to the obstacles set.

  4. canPlaceBlock Method:

    • Parameters: intervals (TreeSet of current intervals), x (upper limit of the range), size (size of the block to place).
    • Iterate through the intervals in intervals:
      • Check if the current interval is large enough to accommodate a block of the given size.
      • Ensure the interval's end does not exceed x by taking Math.min(current.end, x).
    • If a suitable interval is found, return true.
    • If no suitable interval is found, return false.

Main Points:

  • Interval Management:

    • Intervals are managed in a sorted set to allow efficient insertion and deletion.
    • Adding an obstacle splits an interval into two smaller intervals.

  • Query Handling:

    • For a type 1 query ([1, x]), an obstacle is added by updating the intervals.
    • For a type 2 query ([2, x, sz]), the intervals are checked to see if there is a continuous range of length sz within [0, x] that is obstacle-free.

Usage of TreeSet:

  • TreeSet ensures that intervals and obstacles are kept in a sorted order.
  • This allows efficient searching (floor, ceiling) and iteration over the intervals.

This approach leverages the efficient interval management and binary search capabilities of TreeSet to handle the obstacle placement and block fitting queries effectively.

class Solution {
    static class Interval {
        int start, end;
        Interval(int start, int end) {
            this.start = start;
            this.end = end;
        }
    }

    public List<Boolean> getResults(int[][] queries) {
        TreeSet<Interval> intervals = new TreeSet<>((i1, i2) -> {
            int lengthDiff = (i2.end - i2.start) - (i1.end - i1.start);
            return lengthDiff != 0 ? lengthDiff : i1.start - i2.start; 
        });
        intervals.add(new Interval(0, Integer.MAX_VALUE)); // initial interval is all the x axis
        TreeSet<Integer> obstacles = new TreeSet<>();
        List<Boolean> result = new ArrayList<>();
        for (int[] query : queries) {
            if (query[0] == 1) {
                addObstacle(obstacles, intervals, query[1]);
            } else {
                result.add(canPlaceBlock(intervals, query[1], query[2]) ? Boolean.TRUE : Boolean.FALSE);
            }
        }
        return result;
    }

    private void addObstacle(TreeSet<Integer> obstacles, TreeSet<Interval> intervals, int obstacle) {
        Integer lowerBound = obstacles.floor(obstacle);
        if (lowerBound == null) {
            lowerBound = 0;
        }
        Integer upperBound = obstacles.ceiling(obstacle);
        if (upperBound == null) {
            upperBound = Integer.MAX_VALUE;
        }
        // replace one interval with two
        intervals.remove(new Interval(lowerBound, upperBound));
        intervals.add(new Interval(lowerBound, obstacle));
        intervals.add(new Interval(obstacle, upperBound));
        // new obstacle
        obstacles.add(obstacle);
    }

    private boolean canPlaceBlock(TreeSet<Interval> intervals, int x, int size) {
        Interval current = intervals.first();
        while (current != null && current.end - current.start >= size) { // do not consider too short intervals
            if (Math.min(current.end, x) - current.start >= size) { // ensure current.end is not too far right
                return true;
            }
            current = intervals.higher(current);
        }
        return false;
    }
}

Approach

Let's maintain TreeSet with longest free intervals (cuts) on top.
Upon adding obstacle we'll remove a cut with this obstacle from the TreeSet and add two new cuts. To do so we'll need second TreeSet with obstacle coordinates.
When searching for a place for a block we'll start from the longest cut and proceed with checks until current cut is longer than given sz. The check is Math.min(cur.end, x) - cur.start >= sz to ensure the cut is not too far right.

Complexity

n = queries.length
Worst case runtime complexity is O(n2logn) since cur = cuts.higher(cur); is O(logn) and the worst case scenario is to have n / 2 obstacles first and then n / 2 searches with sz = 1 resulting in O(nlogn) for each query.
The more obstacles are added, the less average cut size will be thus when searching for high sz we'll have to iterate through less cuts. So the best-case scenario for single search is under O(nlogn).

  • Space complexity: O(n)

class Solution {
    static class Pair {
        int start, end;
        Pair(int s, int e) {
            start = s;
            end = e;
        }
    }

    public List<Boolean> getResults(int[][] queries) {
        TreeSet<Pair> cuts = new TreeSet<>((p1, p2) -> {
            int diffLen = (p2.end - p2.start) - (p1.end - p1.start);
            return diffLen != 0 ? diffLen : p1.start - p2.start; 
        });
        cuts.add(new Pair(0, Integer.MAX_VALUE)); // initial cut is all the x axis
        TreeSet<Integer> obstacles = new TreeSet<>();
        List<Boolean> result = new ArrayList<>();
        for (int i = 0; i < queries.length; i++) {
            if (queries[i][0] == 1) {
                int x = queries[i][1];
                Integer floor = obstacles.floor(x);
                if (floor == null) {
                    floor = 0;
                }
                Integer ceil = obstacles.ceiling(x);
                if (ceil == null) {
                    ceil = Integer.MAX_VALUE;
                }
                // replace one cut with two
                cuts.remove(new Pair(floor,  ceil));
                cuts.add(new Pair(floor, x));
                cuts.add(new Pair(x, ceil));
                // new obstacle
                obstacles.add(x);
            } else {
                int x = queries[i][1];
                int sz = queries[i][2];
                Pair cur = cuts.first();
                boolean success = false;
                while (cur != null && cur.end - cur.start >= sz) { // do not consider too short cuts
                    if (Math.min(cur.end, x) - cur.start >= sz) { // ensure cur.end is not too far right
                        success = true;
                        result.add(true);
                        break;
                    }
                    cur = cuts.higher(cur);
                }
                if (!success) {
                    result.add(false);
                }
            }
        }
        return result;
    }
}

class Solution {
    public List<Boolean> getResults(int[][] queries) {
        NavigableMap<Integer, Integer> startEnd = new TreeMap<>();
        startEnd.put(0, 50000);
        NavigableMap<Integer, NavigableSet<Integer>> lenStart = new TreeMap<>();
        NavigableSet<Integer> set = new TreeSet<>();
        set.add(0);
        lenStart.put(50000, set);
        List<Boolean> result = new ArrayList<>();
        for (int[] query : queries) {
            if (query[0] == 1) {
                int x = query[1];
                int prev = startEnd.floorKey(x);
                int end = startEnd.get(prev);
                startEnd.put(prev, x);
                startEnd.put(x, end);
                NavigableSet<Integer> starts = lenStart.get(end - prev);
                if (starts.size() == 1) {
                    lenStart.remove(end - prev);
                } else {
                    starts.remove(prev);
                }
                lenStart.computeIfAbsent(x - prev, a -> new TreeSet<>()).add(prev);
                lenStart.computeIfAbsent(end - x, a -> new TreeSet<>()).add(x);
            } else {
                int x = query[1];
                int sz = query[2];
                Integer size = lenStart.ceilingKey(sz);
                boolean found = false;
                while (size != null) {
                    int firstStart = lenStart.get(size).first();
                    if (x - firstStart >= sz) {
                        found = true;
                        result.add(true);
                        break;
                    }
                    size = lenStart.higherKey(size);
                }
                if (!found) {
                    result.add(false);
                }
            }
        }
        return result;
    }
}

This code provides a way to handle and process dynamic range updates and queries. The segment tree structure allows efficient max-range queries and point updates within a range, which is useful in handling large sets of queries with specific constraints.

class Solution {

    class node {
        int val;
        node left, right;
        public node(int val) {
            this.val = val;
        }
    }

    class segmenttree {

        private node root;
        private int size;

        public segmenttree(int n){
            root = create(0, n);
            size = n;
        }

        public void update(int i, int v) {
            update(root, 0, size, i, v);
        }

        public int search(int i) {
            return search(0, i, 0, size, root);
        }

        public int search(int  i, int j, int s, int e, node root) {
            if(s >= i && j >= e) {
                return root.val;
            }
            if(s > j || e < i) {
                return 0;
            }
            int mid = (s+e)/2;
            int left = search(i, j, s, mid, root.left);
            int right = search(i,j, mid+1, e, root.right);
            return Math.max(left, right);
        }

        public void update(node root, int s, int e, int i, int v) {
            if(root == null || e < i || s > i) {
                return;
            }
            if(s == e && s == i) {
                root.val = v;
                return;
            }
            int mid = (s+e)/2;
            update(root.left, s, mid, i, v);
            update(root.right, mid+1, e, i, v);
            root.val = Math.max(root.left != null?root.left.val:0, root.right != null?root.right.val:0);
        }

        public node create(int s, int e) {
            if(s == e) {
                return new node(0);
            }
            int mid = (s+e)/2;
            node temp = new node(0);
            temp.left = create(s, mid);
            temp.right = create(mid+1, e);
            return temp;
        }
    }


    public List<Boolean> getResults(int[][] queries) {
        segmenttree tree = new segmenttree(50001);
        List<Boolean> result = new LinkedList();
        TreeSet<Integer> set = new TreeSet();
        set.add(0);
        set.add(50001);
        tree.update(50001, 50001);
        for(int[] q: queries) {
            if(q[0] == 1) {
                set.add(q[1]);
                tree.update(q[1], q[1] - set.lower(q[1]));
                tree.update(set.higher(q[1]), set.higher(q[1]) - q[1]);
            } else {
               int val = tree.search(set.lower(q[1])); 
               boolean check1 =  val >= q[2];
               boolean check2 = q[1] - set.lower(q[1]) >= q[2];
               result.add(check1 || check2);
            }
        }
        return result;
    }
}
