? Understanding the Problem

  • We're given an n * m array moveTime which represents the time that will be taken to move to a particular room or cell.
  • We can start from room (0, 0) at time t = 0 and we need to find the minimum possible time to reach (n - 1, m - 1), that is, the minimum possible time to reach the bottom right corner of the grid from the top left corner of the grid.
  • Also, moving from one cell to the other costs 1 second and 2 seconds alternately.
  • For example : For moveTime = [[0,4],[4,4]],
  • There are basically 2 possible ways here,

  • And as we can see, it takes equal time to reach the bottom right corner from both the ways so picking any one of the ways is optimal here.
  • If we pick Way 1 then, it takes us
    • 4 seconds to reach (0, 1) + 1 second to move + 2 second to move to (n - 1, m - 1)
    • That is, 4 + 1 + 2 = 7
    • So the minimum time required to reach (n - 1, m - 1) is 7.

class Solution {
public:
    int minTimeToReach(vector<vector<int>>& moveTime) {
        int n = moveTime.size(), m = moveTime[0].size();
        vector<vector<int>> dp(n, vector<int>(m, INT_MAX));
        priority_queue<vector<int>, vector<vector<int>>, greater<vector<int>>> minh;
        minh.push({0, 0, 0});    
        moveTime[0][0] = 0;

        vector<vector<int>> directions = {{1, 0}, {0, 1}, {-1, 0}, {0, -1}};
        while (!minh.empty()) {
            auto current = minh.top();
            minh.pop();
            int currTime = current[0];
            int currRow  = current[1];
            int currCol  = current[2];
            if (currTime >= dp[currRow][currCol]) continue;
            if (currRow == n - 1 && currCol == m - 1) return currTime;
            dp[currRow][currCol] = currTime;

            for (auto& direction : directions) {
                int nextRow = currRow + direction[0];
                int nextCol = currCol + direction[1];
                if (nextRow >= 0 && nextRow < n &&
                    nextCol >= 0 && nextCol < m &&
                    dp[nextRow][nextCol] == INT_MAX) {
                    int cost = (currRow + currCol) % 2 + 1; 
                    int start = max(moveTime[nextRow][nextCol], currTime);
                    int nextTime = start + cost;
                    minh.push({nextTime, nextRow, nextCol});
                }
            }
        }
        return -1;
    }
};

  • And we keep exploring neighbours and add + 1 if the step is even else we add + 2.

class Solution {
public:
    int minTimeToReach(vector<vector<int>>& moveTime) {
        int n = moveTime.size();
        int m = moveTime[0].size();

        vector<pair<int,int>> dir = {{0,1}, {1,0}, {0,-1}, {-1,0}};
        priority_queue<tuple<int,int,int,int>, vector<tuple<int,int,int,int>>, greater<>> minh;
        minh.emplace(0, 0, 0, 0);

        vector<vector<int>> minTime(n, vector<int>(m, INT_MAX));
        minTime[0][0] = 0;

        while (!minh.empty()) {
            auto [currTime, x, y, alt] = minh.top();
            minh.pop();

            if (currTime > minTime[x][y]) continue;
            if (x == n - 1 && y == m - 1) return currTime;

            for (auto [dx, dy] : dir) {
                int nx = x + dx;
                int ny = y + dy;

                if (nx < 0 || nx >= n || ny < 0 || ny >= m) continue;

                int waitTime = max(moveTime[nx][ny] - currTime, 0);
                int moveCost = (alt % 2 == 0 ? 1 : 2);
                int nextTime = currTime + waitTime + moveCost;

                if (nextTime < minTime[nx][ny]) {
                    minTime[nx][ny] = nextTime;
                    minh.emplace(nextTime, nx, ny, alt + 1);
                }
            }
        }
        return -1;
    }
};

class Solution {
  int minTimeToReach(List<List<int>> moveTime) {
    int n = moveTime.length;
    int m = moveTime[0].length;
    List<List<bool>> visited = List.generate(n, (_) => List.filled(m, false));
    
    List<List<int>> bestTime = List.generate(n, (_) => List.filled(m, 1 << 30));
    List<List<int>> dir = [
      [-1, 0],
      [0, 1],
      [1, 0],
      [0, -1]
    ];
    
    int ans = 1 << 30;

    void _helper(int i, int j, int time, int nextMoveTime) {
      if (time >= bestTime[i][j]) return;
      bestTime[i][j] = time;

      if (i == n - 1 && j == m - 1) {
        ans = min(ans, time);
        return;
      }

      for (List<int> d in dir) {
        int x = i + d[0];
        int y = j + d[1];
        if (x >= 0 && x < n && y >= 0 && y < m && !visited[x][y]) {
          visited[x][y] = true;
          int wait = moveTime[x][y];
          int futureMove = nextMoveTime == 1 ?  2 : 1;
          int nextTime = wait > time ? wait + futureMove : time + futureMove;
          if (i == 0 && j == 0 && wait <= time) {
            nextTime = futureMove + wait;
          }
          _helper(x, y, nextTime, futureMove);
          visited[x][y] = false;
        }
      }
    }

    visited[0][0] = true;
    _helper(0, 0, 0,0);
    return ans;
  }
}

? Intuition

To minimize the total time to reach the destination in a grid with dynamic cell availability, we need to prioritize paths with lower travel times, especially when cells have constraints on when they can be visited. This suggests a Dijkstra-like greedy approach.

? Approach

  • Use Dijkstra’s algorithm with a min-heap (PriorityQueue) to always process the cell with the current minimum time.

  • Maintain a bestTime matrix to track the least time to reach each cell and avoid reprocessing worse paths.

  • From each cell, try moving in 4 directions.

    • If the cell isn't yet available (moveTime[x][y] > time), wait until it's open.
    • Use futureMove to alternate movement times between 1 and 2 units (as per the problem's movement constraint).

  • Handle the starting cell specially to account for initial delays.

  • Stop once the destination cell (n-1, m-1) is reached.

Dry Run Example:
Given moveTime = [[0, 2], [1, 3]]:

  • Start at (0,0) with time = 0, futureMove = 1
  • Move to (0,1) ? must wait till 2, then add futureMove = 2 ? time = 4
  • Move to (1,0) ? must wait till 1, then futureMove = 2 ? time = 3
  • Continue to pick minimum from queue until destination reached.

?? Complexity

  • Time complexity: O(n * m * log(n * m))
    Because each cell is pushed to the priority queue with updated time, and heap operations are log(N) where N = n * m.

  • Space complexity: O(n * m)
    For storing bestTime matrix and the priority queue.

import 'package:collection/collection.dart';
import 'dart:math';

class Solution {
  int minTimeToReach(List<List<int>> moveTime) {
    int n = moveTime.length;
    int m = moveTime[0].length;
    List<List<int>> bestTime = List.generate(n, (_) => List.filled(m, 1 << 30));

    List<List<int>> directions = [
      [-1, 0], [0, 1], [1, 0], [0, -1]
    ];

    // Priority queue: stores [time, x, y, nextMove]
    PriorityQueue<List<int>> pq = PriorityQueue((a, b) => a[0] - b[0]);
    pq.add([0, 0, 0, 0]); // time, i, j, nextMoveTime

    while (pq.isNotEmpty) {
      var curr = pq.removeFirst();
      int time = curr[0];
      int i = curr[1];
      int j = curr[2];
      int nextMoveTime = curr[3];

      if (time >= bestTime[i][j]) continue;
      bestTime[i][j] = time;

      if (i == n - 1 && j == m - 1) return time;

      for (var d in directions) {
        int x = i + d[0];
        int y = j + d[1];
        if (x >= 0 && x < n && y >= 0 && y < m) {
          int wait = moveTime[x][y];
          int futureMove = nextMoveTime == 1 ? 2 : 1;
          int nextTime = wait > time ? wait + futureMove : time + futureMove;

          // Special handling for the starting point
          if (i == 0 && j == 0 && wait <= time) {
            nextTime = wait + futureMove;
          }

          if (nextTime < bestTime[x][y]) {
            pq.add([nextTime, x, y, futureMove]);
          }
        }
      }
    }

    return -1; // Should never reach
  }
}
