Intuition

Create the nums array and track the count of adjacent numbers that are same

Approach

  • Create the array nums
  • For each query first check if the values at index - 1 or index + 1 are the same as nums[index]. If any of them are equal, decrement the counter
  • Update the value of nums[index]
  • Now check if values at index - 1 or index + 1 are the same as nums[index]. If any of them are equal, increment the counter

Complexity

  • Time complexity: O(n)

  • Space complexity: O(n)

Code

Python3:

def colorTheArray(self, n: int, queries):
    nums, c = [0] * n, 0
    for index, color in queries:
        pre, nex = nums[index - 1] if index > 0 else 0, nums[index + 1] if index < n-1 else 0
        if nums[index] and nums[index] == pre: c -= 1
        if nums[index] and nums[index] == nex: c -= 1
        nums[index] = color
        if nums[index] == pre: c += 1
        if nums[index] == nex: c += 1
        yield c

class Solution {
    public int[] colorTheArray(int n, int[][] queries) {
      int a[]=new int[n];
      int len=queries.length;
      int res[]=new int[len];
      int count=0;
      for(int i=0;i<len;i++)
      {
          int ind=queries[i][0], color=queries[i][1];
          int prev = ind>0 ? a[ind-1] : 0;
          int next = ind>=n-1 ? 0 : a[ind+1];
          if(a[ind]!=0 && a[ind] == prev)
          {
              count--;
          }
          if(a[ind]!=0 && a[ind] == next)
          {
              count--;
          }
          a[ind]=color;
          if(a[ind] == prev)
          {
              count++;
          }
          if(a[ind] == next)
          {
              count++;
          }
          res[i]=count;
      }
      return res;
    }
}

Intuition

We just need to keep track of the number of adjacent cells with the same color after every query. So for every query we check the 2 neighboring cells of queries[i][0] and update the count factoring in the old color that is going to be removed and the new color that is going to be added. The code should be self-explanatory.

Complexity

  • Time complexity: O(m) where m = queries.length
  • Space complexity: O(n+m)

Code

class Solution {
  public int[] colorTheArray(int n, int[][] queries) {
    var nums = new int[n];
    var m = queries.length;
    var ans = new int[m];
    var cnt = 0;

    for (var i=0; i<m; i++) {
      var j = queries[i][0];

      if (nums[j] != 0) {
        cnt -= j == 0 || nums[j - 1] != nums[j] ? 0 : 1;
        cnt -= j == n-1 || nums[j + 1] != nums[j] ? 0 : 1;
      }
      nums[j] = queries[i][1];

      cnt += j == 0 || nums[j - 1] != nums[j] ? 0 : 1;
      cnt += j == n-1 || nums[j + 1] != nums[j] ? 0 : 1;

      ans[i] = cnt;
    }
    return ans;
  }
}

class Solution {
    public int[] colorTheArray(int n, int[][] queries) {
        int c = 0;
        int[] a = new int[n];
        int[] ans = new int[queries.length];
        for (int i = 0; i < queries.length; i++) {
            int p = queries[i][0];
            int  m = queries[i][1];
            c = update_color(c, a, p, false);
            a[p] = m;
            c = update_color(c, a, p, true);
            ans[i] = c;
        }
        
        return ans;
    }
    public int update_color(int c, int[] a, int p, boolean inc) {
        if (p > 0 && a[p - 1] == a[p] && a[p - 1] != 0) {
            c += inc ? 1 : -1;
        }
        if (p < a.length - 1 && a[p + 1] == a[p] && a[p + 1] != 0) {
            c += inc ? 1 : -1;
        }
        return c;
    }
}

class Solution {
    public int[] colorTheArray(int n, int[][] queries) {
        if(n==1) return new int[queries.length];
        
        int ans[]=new int[queries.length];  // for storing the answer of each query
        int color[]=new int[n];  
        int findPair=0;     // it will keep track of total Number of Pair(adject)

        for(int i=0;i<queries.length;i++){

                int index=queries[i][0];     // index which we need to color
                int assignColor=queries[i][1];  // given color for current index;

// find the adjecent leftColor and rightColor of the current Index
                int leftColor=(index-1)>=0?color[index-1]:0;    
                int rightColor=(index+1)<n?color[index+1]:0;    

// if current index color is forming pair to left or right ,then we need to decrease the findPair by 1 because we are going to 
// color the current index with a new color
                if(color[index]!=0 && leftColor==color[index] ) findPair--;  // if current index color matches with left one
                if(color[index]!=0 && rightColor==color[index]) findPair--;  // if current index color matches with right one

// color the current index with given new Color
                color[index]=assignColor;

// Now check if, after assigning the new Color to the current index ,it is forming any new Pair,if it is  then increase the 
// findPair by 1;
                if(leftColor!=0 && color[index]==leftColor) findPair++;  // if color matches with leftColor
                if(rightColor!=0 && color[index]==rightColor) findPair++; // if color matches with rightColor

        // assign the calculated pair to the answer
                ans[i]=findPair;
        }

        return ans;
    }
}

class Solution {
    public int[] colorTheArray(int n, int[][] queries) {
        int ans = 0;
        int[] colors = new int[n];
        Arrays.fill(colors, 0);
        int[] result = new int[queries.length];
        int i =0;

        for(int[] query : queries) {
            int index = query[0];
            int nextColor = query[1];
            int prevColor = colors[index];
            int beforeColor = 0;
            int afterColor = 0;

            colors[index] = nextColor;

            if (index > 0) {
                beforeColor = colors[index-1];
            }

            if (index < (n-1)) {
                afterColor = colors[index+1];
            }

            if (prevColor != 0) {
                if (beforeColor == prevColor) ans--;
                if (afterColor == prevColor) ans--;
            }
            if (beforeColor == nextColor) ans++;
            if (afterColor == nextColor) ans++;
            result[i++] = ans;
        }
        return result;
    }
}
