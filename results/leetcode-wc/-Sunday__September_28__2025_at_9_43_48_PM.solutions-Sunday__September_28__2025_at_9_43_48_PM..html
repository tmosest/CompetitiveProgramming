Approach

  • Write a function to check the operations for a particular value y and a value of not y cell.
  • Then check the operation count for all the 6 combinations (0,1), (1,0), (1,2), (2,1), (0,2), (2,0) using above function and take the min as the answer
  • To check if the cell is belong to Y or not here are the 3 cases we need to check for :
    • To check for left side diagonal of Y, check if both row & col are equal or not and the row is belong to upper half of the grid => (i <= n/2 && i == j )
    • To check for the right side diagonal of Y, check if the sum of row + col is equal to grid.size()-1 and it must be on upper half of the rows => (i <= n/2 && i + j == n - 1)
    • To check the vertical line of the Y we need to check if the row is in lower half and col is equal to half of the total col size => (i > n/2 && j == n/2)

Complexity

  • Time complexity: O(N.N)
  • Space complexity: O(1)

C++

int getOperationCount(vector<vector<int>>& grid, int y, int notY){
    int ans = 0;
    for(int i = 0; i < grid.size(); ++i){
        for(int j = 0; j < grid[0].size(); ++j){
            if((i <= grid.size()/2 && (i == j || i + j == grid.size() - 1)) || (i > grid.size()/2 && j == grid.size()/2)){
                if(y != grid[i][j]) ans++;
            }else{
                if(notY != grid[i][j]) ans++;
            }
        }
    }
    return ans;
}

int minimumOperationsToWriteY(vector<vector<int>>& grid) {
    return min({getOperationCount(grid, 0,1), getOperationCount(grid, 1,0), getOperationCount(grid, 1,2), getOperationCount(grid, 2,1), getOperationCount(grid, 2,0), getOperationCount(grid, 0,2)});
}

int getOperationCount(int[][] grid, int y, int notY){
    int ans = 0, n = grid.length;
    for(int i = 0; i < n; ++i){
        for(int j = 0; j < n; ++j){
            if((i <= n/2 && (i == j || i + j == n - 1)) || (i > n/2 && j == n/2)){
                if(y != grid[i][j]) ans++;
            }else{
                if(notY != grid[i][j]) ans++;
            }
        }
    }
    return ans;
}
    
public int minimumOperationsToWriteY(int[][] grid) {
    int ans = Integer.MAX_VALUE;
    for(int i = 0; i < 3; ++i){
        for(int j = 0; j < 3; ++j){
            if(i != j) ans = Math.min(ans, getOperationCount(grid, i,j));
        }
    }
    return ans;
}

Intuition

There can be at most 6 combination n * (n-1). We can either try out all combination and traverse the grid 6 times or we can store frequency of elements for Y and non-Y in one iteration and identify the change cost from the frequency.

Approach

First traverse through grid and add frequency of numbers 0,1,or 2 for Y and non-Y. Also, store how many numbers are there for each of them.

Then, try those 6 combination. Let's say to convert value to 0, our cost will be total - freq[0]. Use this for both Y and non-Y and you get your total cost.

Complexity

  • Time complexity: O(n*n)
  • Space complexity: O(1)

Code

class Solution {
    public int minimumOperationsToWriteY(int[][] grid) {
        int n = grid.length;
        int center = n/2;

        int[] yFreq = new int[3]; // Store frequency of 0, 1, and 2
        int[] nonYFreq = new int[3];
        int totalY = 0; // Total elements on Y.
        int totalNonY = 0;

        for(int i=0; i<n; i++) {
            for(int j=0; j<n; j++) {
                if(isOnY(n,center,i,j)) {
                    yFreq[grid[i][j]]++;
                    totalY++;
                } else {
                    nonYFreq[grid[i][j]]++;
                    totalNonY++;
                }
            }
        }

        // Now that we have freq of all, we can just try differetn combination of y and non-y between 0,1,2.
        int minOp = Integer.MAX_VALUE;
        for(int y = 0; y < 3; y++) {
            for(int nonY = 0; nonY < 3; nonY++) {
                if(y == nonY)
                    continue;
                int yModifyCost = totalY - yFreq[y]; //Cost of changing all items on Y to y
                int nonYModifyCost = totalNonY - nonYFreq[nonY]; //Cost of changing all items on NonY to nonY
                minOp = Math.min(minOp, yModifyCost + nonYModifyCost);
            }
        }
        return minOp;
    }

    public boolean isOnY(int n, int center, int r, int c) {
        return (r==c && r<=center ) || ( (r+c+1) == n && r<=center) || (r > center && c == center);
    }
}
