//Approach-1 (Brute Force)
//T.C : O(m*n)
//S.C : O(1)
class Solution {
public:
    int countMatchingSubarrays(vector<int>& nums, vector<int>& pattern) {
        int n = nums.size();
        int m = pattern.size();
        
        int count = 0;
        
        for(int i = 0; i < n-m; i++) {

            int k = 0;
            while(k < m) {
                if(pattern[k] == 1 && nums[i+k+1] > nums[i+k]) {
                    k++;
                } else if(pattern[k] == 0 && nums[i+k+1] == nums[i+k]) {
                    k++;
                } else if(pattern[k] == -1 && nums[i+k+1] < nums[i+k]) {
                    k++;
                } else {
                    break;
                } 
            }

            if(k == m) { //Found complete pattern
                count++;
            }
            
        }
        
        return count;
    }
};

//T.C : Same as KMP O(m+n)
//S.C : O(m)
class Solution {
public:
    void longestPrefixSuffix(vector<int>& pattern, vector<int>& lps) {
        int M = pattern.size();
        int len = 0; // Length of the previous longest prefix & suffix
    
        lps[0] = 0; // Because there is no proper suffix and prefix of pattern[0..0]
    
        int i = 1;
        while (i < M) {
            if (pattern[i] == pattern[len]) {
                len++;
                lps[i] = len;
                i++;
            } else {
                if (len != 0) {
                    len = lps[len - 1]; //You can also write, len = len-1;
                } else {
                    lps[i] = 0;
                    i++;
                }
            }
        }
    }

    int countMatchingSubarrays(vector<int>& nums, vector<int>& pattern) {
        int N = nums.size();
        int M = pattern.size();

        vector<int> nums2;
        for(int i = 0 ; i < N-1 ; i++) {
            if(nums[i] < nums[i+1]) 
                nums2.push_back(1);
            else if(nums[i] > nums[i+1]) 
                nums2.push_back(-1);
            else 
                nums2.push_back(0);
        }
        
        vector<int> lps(M, 0);
        longestPrefixSuffix(pattern, lps);

        int i       = 0;
        int j       = 0;
        int result  = 0;

        while(i < nums2.size()) {
            if(pattern[j] == nums2[i]) {
                i++;
                j++;
            }
            
            if (j == pattern.size()) {
                result++; //Pattern found at index i-j+1 (If you have to return 1 Based indexing, that's why added + 1)
                j = lps[j - 1];
            } else if (i < nums2.size() && pattern[j] != nums2[i]) {
                if (j != 0) {
                    j = lps[j - 1];
                } else {
                    i++;
                }
            }

        }
        return result;


    }
};

Intuition

The key insight is that we can use a rolling hash algorithm to efficiently check for matching subarrays. Instead of comparing each subarray element by element, we can convert the pattern and subarrays into a hash value and compare those hashes. This is particularly efficient for large arrays.

The solution transforms both the pattern and the sliding window of nums into a sequence of -1, 0, and 1 (shifted by +1 to be 0, 1, 2 for better hash computation) based on the comparison between adjacent elements.

Approach

  1. Hash Function

    • We implement a custom hash function that computes a rolling hash using BASE = 3 (since we have 3 possible values: 0, 1, 2)
    • The hash is computed as: h = (h * BASE + value) % MOD for each value in the sequence
    • We use a large prime MOD to minimize hash collisions

  2. Pattern Processing

    • Convert the pattern array into a sequence of numbers by adding 1 to each value
    • Compute the target hash value for this pattern

  3. Computing Base Power

    • Calculate BASE^(n-1) % MOD which is needed for rolling hash updates
    • This value is used to remove the leftmost element's contribution when sliding the window

  4. Array Processing

    • Convert the input array into a sequence of comparisons between adjacent elements
    • For each pair (a, b):
      • If b > a, store 2 (original 1 + 1)
      • If b == a, store 1 (original 0 + 1)
      • If b < a, store 0 (original -1 + 1)

  5. Rolling Hash Window

    • Compute the initial hash for the first window
    • For each subsequent window:
      • Remove the leftmost element: hash = hash - leftmost * BASE^(n-1)
      • Shift left: hash = hash * BASE
      • Add new rightmost element: hash = hash + new_element
      • All operations are done modulo MOD
    • Increment answer when current hash matches target hash

Complexity

  • Time Complexity: O(n) where n is the length of nums

    • Converting nums to hills array: O(n)
    • Computing initial hash: O(m) where m is pattern length
    • Rolling hash window: O(n)

  • Space Complexity: O(n)

    • We store the hills array which has length n-1
    • Other variables use constant space

The rolling hash approach is much more efficient than naive pattern matching which would require O(n*m) comparisons. By using hashing, we reduce the comparison to O(1) for each window.

class Solution {
    private fun List<Int>.hash(range: IntRange = 0..lastIndex): Long {
        var h = 0L
        for (i in range) {
            h = (h * BASE % MOD + this[i]) % MOD
        }
        return h
    }

    fun countMatchingSubarrays(nums: IntArray, pattern: IntArray): Int {
        val n = pattern.size
        val targetHash = pattern.map { it + 1 }.hash()
        val basePow = (0..<(n - 1)).fold(1L) { acc, _ -> acc * BASE % MOD }

        val hills = nums.toList().zipWithNext().map { (a, b) -> (b compareTo a) + 1 }
        var currentHash = hills.hash(0..<n)

        var ans = if (targetHash == currentHash) 1 else 0
        for (i in n..hills.lastIndex) {
            currentHash = ((currentHash + MOD - hills[i - n] * basePow % MOD) % MOD * BASE % MOD + hills[i]) % MOD
            ans += if (targetHash == currentHash) 1 else 0
        }

        return ans
    }

    companion object {
        const val BASE = 3L
        const val MOD = 1_000_000_000_000_000 - 11
    }
}
