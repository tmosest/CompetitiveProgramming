class TrieNode {
        public :
        TrieNode *ch[26];
        bool isEnd;
        TrieNode() {
            for(int i=0;i<=25;i++)
                ch[i]=NULL;
            isEnd=false;
        }
    };
class Solution {
public:
    int dp[10001];
    

    class Trie {
        public:
        TrieNode *root;
        Trie() {
            root=new TrieNode();
        }
        TrieNode* getRoot() {
            return root;
        }
        void insert(string &s) {
            int n = s.size();
            TrieNode *t=root;
            for(int i=0;i<n;i++) {
                int k = s[i]-'a';
                if(t->ch[k]==NULL)
                    t->ch[k]= new TrieNode();
                t=t->ch[k];
            }
            t->isEnd=true;
        }
        int search(string &s,int i,int n,int dp[]) {
            if(i==n) {
                return 0;
            }
            if(dp[i]!=-1)
                return dp[i];
            int ans=1e6;
            TrieNode *h =root;
            for(int j=i;j<n;j++) {
                int k = s[j]-'a';
                if(h->ch[k]!=NULL) {
                    ans=min(ans,1+search(s,j+1,n,dp));
                    h=h->ch[k];
                } else
                    break;
            }
            return dp[i] = ans;
        }
    };

    int minValidStrings(vector<string>& words, string s) {
        Trie trie;
        for(auto i:words) {
            trie.insert(i);
        }   
        int n = s.size();
        memset(dp,-1,sizeof(dp));
        int k = trie.search(s,0,n,dp);
        if(k>1e4)
            return -1;
        return k;
    }
};

Intuition

  1. To solve the problem of determining the minimum number of valid strings required to construct a target string from a given list of words, we need to:

  2. Understand Substring Matching: Determine if substrings from each word can be matched to segments of the target string.
    Optimize for Minimum Strings: Use dynamic programming or sliding window techniques to efficiently find the minimum number of words needed.

Approach

Approach Overview

  1. Hashing for Substring Matching:
  • Use polynomial rolling hashing to efficiently compute and compare substring hashes. This will allow quick checks if a substring from a word matches a segment of the target string.
    Dynamic Programming (DP)
  1. Setup:
  • Create a DP array where dp[i] represents the minimum number of words required to form the substring of the target from the start up to position i.
  • Initialize dp[0] to 0 (no words needed to form an empty substring) and other entries to infinity (INF).
  1. DP Transition:
  • For each position i in the target string, try to extend the substring using each word from the list.
  • Use binary search to find the longest valid prefix from each word that matches a segment of the target string.
  • Update the DP array based on whether extending with the current word improves the solution.
  1. Result Extraction:
  • The value in dp[target.length()] will give the minimum number of words needed to construct the entire target string.

Detailed Steps

  1. Pre-compute Hashes:
  • Compute hash values for all substrings of each word and the target string.
  1. Initialize DP Table:
  • Set dp[0] = 0 and initialize all other entries to INF.
  1. Update DP Table:
  • For each position i in the target string, check if appending each word can form a valid substring, and update the DP array.
  1. Extract Result:
  • The result will be dp[target.length()], which represents the minimum number of words required to form the target string. If it remains INF, it means forming the target string is not possible with the given words.

Example

If the target is "abc" and the words are ["a", "b", "c"], the DP array would update as follows:

  • Start with dp[0] = 0.
  • Process each word and update the DP values for each valid extension.
  • Finally, dp[3] (where the length of "abc" is 3) will give the answer.

Complexity

Time complexity:

  • Hash computation and substring comparisons take O(NáM) where
    N is the length of the target and M is the length of the longest word.
  • DP updates involve scanning through all positions and words, which results in a complexity of O(N*W*logM) where W is the number of words.
  • Overall time complexity: O(N*W*logM), considering efficient hash operations and binary search.

Space complexity:

  • Hash Storage: O(N+W*M), to store hash values for the target and all words.
  • DP Table: O(N), to store the minimum words needed for each position in the target string.
  • Auxiliary Structures: Additional space for data structures like TreeMap and HashMap.

class Solution {
    public int minValidStrings(String[] words, String target) {
        int w[] = new int[50001], h[][] = new int[words.length + 1][50001], result = -1;
        w[0] = 1;
        for (int i = 1; i <= 50000; i++) {
            w[i] = w[i - 1] * 31;
        }
        for (int i = 0; i <= words.length; i++) {
            for (int j = 0; j < (i < words.length ? words[i] : target).length(); j++) {
                h[i][j + 1] = h[i][j] + (i < words.length ? words[i] : target).charAt(j) * w[j];
            }
        }
        TreeMap<Integer, Integer> tree = new TreeMap<>(Map.of(0, 1));
        HashMap<Integer, ArrayList<Integer>> hash = new HashMap<>(Map.of(1, new ArrayList<>(List.of(0))));
        for (int i = 0; i <= target.length(); i++) {
            for (int v : hash.getOrDefault(i, new ArrayList<>())) {
                tree.put(v, tree.get(v) - 1);
                if (tree.get(v) == 0) {
                    tree.remove(v);
                }
            }
            result = tree.isEmpty() ? -1 : tree.firstKey();
            if (!tree.isEmpty()) {
                int left = 0;
                for (int j = 0; j < words.length; j++) {
                    for (int right = Math.min(words[j].length(), target.length() - i); left < right;) {
                        int mid = (left + right + 1) / 2;
                        if (h[words.length][mid + i] - h[words.length][i] == h[j][mid] * w[i]) {
                            left = mid;
                        } else {
                            right = mid - 1;
                        }
                    }
                }
                if (left > 0) {
                    tree.put(result + 1, tree.getOrDefault(result + 1, 0) + 1);
                    hash.computeIfAbsent(i + left + 1, t -> new ArrayList<>()).add(result + 1);
                }
            }
        }
        return result;
    }
}
