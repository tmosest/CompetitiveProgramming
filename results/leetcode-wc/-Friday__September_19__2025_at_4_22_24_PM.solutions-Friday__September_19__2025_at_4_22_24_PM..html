DFS

Approach: For each initial node, do a DFS of the graph and count how many nodes would get saved if this node was not there.
Optimization 1: Since removing this node can only affect the fate of nodes in its connected component, do a DFS only over nodes connected to this node.
Optimization 2: If we encounter another initial node during a DFS, then we mark all the nodes in the path as infected too. Remember, we started from an infected node and now we have encounted another infected node so this means that no matter what, the nodes in between are doomed to get infected eventually (removing any one of the initial nodes isn't going to save them).

// 2 ms. 92.26%
class Solution {
    private int dfs(int[][] graph, int u, boolean[] visited, boolean[] infected) {
        if(infected[u]) return 0;
        visited[u] = true;
        int count = 1;
        for(int v = 0; v < graph[u].length; v++) {
            if(!visited[v] && graph[u][v] == 1) {
                int c = dfs(graph, v, visited, infected);
                if(c == 0) {
                    infected[u] = true;
                    return 0;
                }
                count += c;
            }
        }
        return count;
    }
    public int minMalwareSpread(int[][] graph, int[] initial) {
        int n = graph.length, ans = initial[0], max = 0;
        boolean[] infected = new boolean[n];
        for(int u: initial) infected[u] = true;
        for(int u: initial) {
            boolean[] visited = new boolean[n];
            visited[u] = true;
            int count = 0;
            for(int i = 0; i < n; i++) {
                if(!visited[i] && graph[u][i] == 1) {
                    count += dfs(graph, i, visited, infected);
                }
            }
            if(count > max || count == max && u < ans) {
                max = count;
                ans = u;
            }
        }
        return ans;
    }
}

Time complexity: O(K N2) where N = number of total nodes and K = number of initially infected nodes.
Space complexity: O(N)

Disjoint Set / Union Find

Approach:
First we identify the connected components in the graph without considering infected nodes. Then we consider each infected node one by one and check how many components are connected to it if we were to add this node (and its edges) into the graph. After having processed all the nodes, we iterate over infected nodes again to see which connected components are exclusively affected by a single infected node because that's the component which'll get saved on removing that infected node. (If a component is connected to multiple infected nodes, removing only one of the infected nodes is not going to save it.)

Union-Find or Disjoint-Set is a perfect data structure for our use case of efficiently adding nodes (and their edges) one by one and to keep track of connected components and their sizes.

// 6 ms. 66.45%
class Solution {
    class UnionFind {
        public int[] parent, size;
        UnionFind(int n) {
            parent = new int[n];
            size = new int[n];
            for(int i = 0; i < n; i++) {
                parent[i] = i;
                size[i] = 1;
            }
        }
        public int find(int x) {
            return x != parent[x] ? x = find(parent[x]) : x;
        }
        public void union(int x, int y) {
            int a = find(x), b = find(y);
            if(a != b) {
                if(size[a] < size[b]) {
                    parent[a] = b;
                    size[b] += size[a];
                } else {
                    parent[b] = a;
                    size[a] += size[b];
                }
            }
        }
    }
    public int minMalwareSpread(int[][] graph, int[] initial) {
        int n = graph.length, ans = initial[0], max = 0;
        boolean[] infected = new boolean[n];
        for(int u: initial) infected[u] = true;
        UnionFind uf = new UnionFind(n);
        for(int i = 0; i < n; i++) {
            if(!infected[i]) {
                for(int j = i + 1; j < n; j++) {
                    if(!infected[j] && graph[i][j] == 1) uf.union(i, j);
                }
            }
        }
        int[] count = new int[n];
        Set<Integer>[] component = new Set[n];
        for(int u: initial) {
            component[u] = new HashSet<>();
            for(int v = 0; v < n; v++) {
                if(!infected[v] && graph[u][v] == 1) component[u].add(uf.find(v));
            }
            for(int v: component[u]) count[v]++;
        }
        for(int u: initial) {
            int save = 0;
            for(int v: component[u]) {
                if(count[v] == 1) save += uf.size[v];
            }
            if(save > max || save == max && u < ans) {
                max = save;
                ans = u;
            }
        }
        return ans;
    }
}

Time complexity: O(N2)
Space complexity: O(N)

Tarjan's algorithm

Approach: We can apply Tarjan's algorithm to find all the articulation points in the graph in O(V + E) time via a DFS. An articulation point is a node removing which disconnects the graph. At the time of identifying whether a node is an articulation point or not, the algorithm also (automatically?) tells us precisely which DFS subtree (or biconnected component) would get disconnected from the graph if we were to remove this node. We can use this information to solve our problem i.e. if that subtree does not contain any other initially infected nodes, then it is clear that removing this node is going to save that entire subtree otherwise it doesn't matter, that subtree is doomed anyway. So all we need to do extra (apart from the usual stuff for Tarjan's algo) is to make DFS method return the size of the connected component when there are no infected nodes in it otherwise 0.

// 2 ms. 92.26%
class Solution {
    private int dfs(int[][] graph, int u, int parent, int time, int[] depth, int[] low, boolean[] infected, int[] count) {
        low[u] = depth[u] = time;
        boolean flag = infected[u];
        int size = 1;
        for(int v = 0; v < graph[u].length; v++) {
            if(graph[u][v] == 1) {
                if(depth[v] == 0) {
                    int s = dfs(graph, v, u, time + 1, depth, low, infected, count);
                    if(s == 0) {
                        flag = true;
                    } else {
                        size += s;
                    }
                    if(low[v] >= depth[u]) {
                        count[u] += s;
                    }
                    low[u] = Math.min(low[u], low[v]);
                } else if(v != parent) {
                    low[u] = Math.min(low[u], depth[v]);
                }
            }
        }
        return flag ? 0 : size;
    }
    public int minMalwareSpread(int[][] graph, int[] initial) {
        int n = graph.length, ans = initial[0], max = 0;
        boolean[] infected = new boolean[n];
        for(int u: initial) infected[u] = true;
        int[] depth = new int[n], low = new int[n], count = new int[n];
        for(int u: initial) {
            if(depth[u] == 0) {
                dfs(graph, u, -1, 1, depth, low, infected, count);
            }
            if(count[u] > max || count[u] == max && u < ans) {
                max = count[u];
                ans = u;
            }
        }
        return ans;
    }
}

Time complexity: O(N2)
Space complexity: O(N)

Intuition

The malware will spread to every nodes if an infected node is present in that particular component. So we need to check that after removing which infected node will save maximum non-infected

Approach

Union Find approach

  • For each infected node we will build a connected components without including that particular node in the graph.
  • After that we will check of nodes that will remain non-infected after removing that particular infected node.
  • Now we will check the same for all the other infected nodes and will return that particular node that will save maximum non-infected node on removing it.
  • If any two nodes are saving same number of non-infected nodes then we will store the smallest node

Complexity

  • Time complexity:

    • For each candidate removal (there are k = initial.size()):
    • We rebuild the DSU in O(n^2) (since we scan the adjacency matrix).
    • Then we check infections in O(n).
      Total: O(k á (n^2 + n)) = O(k á n^2)

  • Space complexity: O(n)

#include <bits/stdc++.h>
using namespace std;

struct DSU {
    vector<int> parent, sz;
    DSU(int n) : parent(n), sz(n, 1) {
        iota(parent.begin(), parent.end(), 0);
    }
    int find(int x) {
        return parent[x] == x ? x : parent[x] = find(parent[x]);
    }
    void unite(int a, int b) {
        a = find(a); b = find(b);
        if (a == b) return;
        if (sz[a] < sz[b]) swap(a, b);
        parent[b] = a;
        sz[a] += sz[b];
    }
    int compSize(int x) { return sz[find(x)]; }
};

class Solution {
public:
    int minMalwareSpread(vector<vector<int>>& graph, vector<int>& initial) {
        int n = graph.size();
        sort(initial.begin(), initial.end()); 

        int bestNode = initial[0];
        int bestCount = INT_MAX;

        for (int rem : initial) {
            DSU dsu(n);
            for (int i = 0; i < n; i++) {
                if (i == rem) continue;
                for (int j = i + 1; j < n; j++) {
                    if (j == rem) continue;
                    if (graph[i][j]) dsu.unite(i, j);
                }
            }

            vector<int> vis(n, 0);
            int total = 0;
            for (int node : initial) {
                if (node == rem) continue;
                int root = dsu.find(node);
                if (vis[root]) continue;
                vis[root] = 1;
                total += dsu.compSize(root);
            }

            if (total < bestCount || (total == bestCount && rem < bestNode)) {
                bestCount = total;
                bestNode = rem;
            }
        }

        return bestNode;
    }
};
