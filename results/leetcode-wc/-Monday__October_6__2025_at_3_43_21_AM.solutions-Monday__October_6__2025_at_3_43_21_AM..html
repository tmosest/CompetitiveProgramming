        Integer startMM = Integer.parseInt(startTime.substring(3));
        Integer finishHH = Integer.parseInt(finishTime.substring(0,2));
        Integer finishMM = Integer.parseInt(finishTime.substring(3));

        int end = finishHH*60 +finishMM;
        if (start > end) {
           end += 24*60;
        }
        
        return (int) Math.floor(end / 15.00) - (int) Math.ceil(start / 15.00);
    }
}

Here we're basically converting the startTime and endTime to minutes and then calculating then result.

class Solution {
    public int numberOfRounds(String startTime, String finishTime) {
        
        int finMin = Integer.parseInt(finishTime.substring(3, 5));
        int startMin = Integer.parseInt(startTime.substring(3,5));
        int finHr = Integer.parseInt(finishTime.substring(0, 2));
        int startHr = Integer.parseInt(startTime.substring(0, 2));
        
        int start = startHr * 60 + startMin; //converting start time to minutes
        int end = finHr * 60 + finMin;  // converting endtime to minutes
        
        if(start % 15 != 0) {     // making start to next multiple of 15 
            int div = start/15;   // doing this because in start time for example of 03:10(190 mins), 
            start = 15 * (div + 1);  //we can join the contest at 03:15(195 mins) only and 195 is next multiple of 15 after 190.
        }
        
        if(end % 15 != 0) {  //doing the same here but instead of finding the next multiple we're rounding off to previous multiple of 15
            int div = end/15; // as ending time of for example 03:10 we have to count till 03:00 because at 03:00 new game starts
            end = 15 * div; // but it ends at 03:15 but our end time is 03:10
        }
        
        if(end < start) end = end + 24*60;  //here if ending time < start time. see example 2 of the provided test cases.
                                            // basically the game ends at next day so we added 24 more hours
        
        
        return Math.abs(end - start) / 15; // now to find the number of games played we divide the time difference by 15
    }
}

class Solution {
    public int numberOfRounds(String startTime, String finishTime) {
        int start = timeInMinute(startTime); //convert start time to minutes
        int end = timeInMinute(finishTime); //Convert end time to minutes
        
        if(end < start)
            end += 24*60;
        
        if(start % 15 != 0)
            start += 15 - (start % 15); // make the starting point from the multiple of 15
        
        if(end % 15 != 0)
            end -= end % 15; // make the ending point multiple of 15
        
        if(end < start)
            return 0;
        
        return (end-start)/15;
        
    }
    
    private int timeInMinute(String s){
        int h = Integer.parseInt(s.substring(0,2));
        int m = Integer.parseInt(s.substring(3));
        
        return (h*60)+m;
    }
}
