Approach

  1. State Definition: Define the state of DP with two parameters:

    • Current Index: represents the position in the arrays where the decision is being made.
    • Last Choice: indicates whether the last chosen energy boost was from array A or array B.

  2. State Representation: Use a DP table where dp[i][curr] represents the maximum energy boost achievable starting from index i with curr indicating the last choice:

    • curr = 1: The last boost was chosen from array A.
    • curr = 0: The last boost was chosen from array B.

  3. Transition: At each index:

    • If curr is 1 (last choice was A): Compute the maximum boost by either:
      • Continuing with array A (adding the current element from A).
      • Switching to array B.
    • If curr is 0 (last choice was B): Compute the maximum boost by either:
      • Continuing with array B (adding the current element from B).
      • Switching to array A.

  4. Finally compute the result by starting from index 0 and considering both possibilities (starting with A or B).

class Solution {
public:
    #define ll long long
    vector<vector<ll>> dp;
    ll f(vector<int>& A, vector<int>& B, int i, int curr) {
        if (i >= A.size()) return 0;
        if(dp[i][curr] != -1) return dp[i][curr];

        ll ans = 0;
        if(curr)
            ans = max(A[i] + f(A, B, i + 1, curr), f(A, B, i + 1, curr ^ 1));

        else
            ans = max(B[i] + f(A, B, i + 1, curr), f(A, B, i + 1, curr ^ 1));

        return dp[i][curr] = ans;
    }

    long long maxEnergyBoost(vector<int>& A, vector<int>& B){
        int n = A.size();
        dp.clear();
        dp.resize(n, vector<ll>(2, -1));  
        return max(f(A, B, 0, 1), f(A, B, 0, 0));
    }
};

class Solution {
public:
    #define ll long long

    long long maxEnergyBoost(vector<int>& A, vector<int>& B) {
        int n = A.size();
        vector<vector<ll>> dp(n + 5, vector<ll>(2, 0));

        for(int i = n - 1; i >= 0; i--){ 
            dp[i][0] = max(B[i] + dp[i + 1][0], dp[i + 1][1]);
            dp[i][1] = max(A[i] + dp[i + 1][1], dp[i + 1][0]);
        }

        return max(dp[0][0], dp[0][1]);
    }
};

class Solution {
public:
    long long maxEnergyBoost(vector<int>& A, vector<int>& B) {
        int n = A.size();
        vector<long long> curr(2, 0), next(2, 0);

        for(int i = n - 1; i >= 0; i--){
            curr[0] = max(B[i] + next[0], next[1]);
            curr[1] = max(A[i] + next[1], next[0]);
            next = curr;
        }

        return max(next[0], next[1]);
    }
};

class Solution {
    //Considering 0 indexed hours
    //drinkType = 0 ===> drink1
    //drinkType = 1 ===> drink2
    long solve(int[] drink1, int[] drink2, int currHour, int currDrinkType)
    {
        if (currHour >= drink1.length)
            return 0;
        
        long drinkBoost = (currDrinkType == 0)? drink1[currHour] : drink2[currHour];
        
        long noTypeSwitch = drinkBoost + solve(drink1, drink2, currHour + 1, currDrinkType);
        long typeSwitch = drinkBoost + solve(drink1, drink2, currHour + 2, 1 - currDrinkType);
        //if you switch types you need to skip next hour so (currHour + 2)
        //(1 - currDrinkType) switches the type (0 to 1) and (1 to 0)
        
        return Math.max(noTypeSwitch, typeSwitch);
        
    }
    public long maxEnergyBoost(int[] energyDrinkA, int[] energyDrinkB) 
    {
        long a = solve(energyDrinkA, energyDrinkB, 0, 0);
        long b = solve(energyDrinkA, energyDrinkB, 0, 1);
        
        return Math.max(a, b);
    }
}

class Solution {
    //Considering 0 indexed hours
    //drinkType = 0 ===> drink1
    //drinkType = 1 ===> drink2
    public long maxEnergyBoost(int[] drink1, int[] drink2) 
    {
        int n = drink1.length;
		
		 //dp[drinkType][hour]
        long[][] dp = new long[2][n];
        
         //in the last hour, for any given type we have only one option i.e to "DRINK"
        dp[0][n - 1] = drink1[n - 1];
        dp[1][n - 1] = drink2[n - 1];

        for (int currHour = n - 2; currHour >= 0; currHour--)
        {
            for (int currDrinkType = 0; currDrinkType <= 1; currDrinkType++)
            {
                long drinkBoost = (currDrinkType == 0)? drink1[currHour] : drink2[currHour];
                
                long noTypeSwitch = drinkBoost + dp[currDrinkType][currHour + 1];
                long typeSwitch = drinkBoost + ( (currHour + 2 >= n)? 0 : dp[1 - currDrinkType][currHour + 2] );
                
                dp[currDrinkType][currHour] = Math.max(noTypeSwitch, typeSwitch);
            }
        }
        return Math.max(dp[0][0], dp[1][0]);
        
    }
}

class Solution {
public:
    int n;
    long long dp[100001][2];
    long long solve(int i,vector<int>& v1, vector<int>& v2,int f) {
        if(i>=n)
            return 0;
        if(dp[i][f]!=-1)
            return dp[i][f];
        if(f==0) {
            return dp[i][f]=v1[i]+max(solve(i+1,v1,v2,0),solve(i+2,v1,v2,1));
        } 
        return dp[i][f]=v2[i]+max(solve(i+2,v1,v2,0),solve(i+1,v1,v2,1));
    }
    long long maxEnergyBoost(vector<int>& v1, vector<int>& v2) {
        n=v1.size();
        memset(dp,-1,sizeof(dp));
        long long ans1=solve(0,v1,v2,0);
        memset(dp,-1,sizeof(dp));
        long long ans2=solve(0,v1,v2,1);
        return max(ans1,ans2);
    }
};

class Solution {
public:
    long long maxEnergyBoost(vector<int>& v1, vector<int>& v2) {
        int n=v1.size();
        long long dp[n+1][2];
        memset(dp,-1,sizeof(dp));
        dp[0][0]=v1[0];
        dp[0][1]=v2[0];
        for(int i=1;i<n;i++) {
            dp[i][0]=v1[i]+dp[i-1][0];
            dp[i][1]=v2[i]+dp[i-1][1];
            if(i-2>=0) {
                dp[i][0]=max(dp[i][0],v1[i]+dp[i-2][1]);
                dp[i][1]=max(dp[i][1],v2[i]+dp[i-2][0]);
            }
        }
        return max(dp[n-1][0],dp[n-1][1]);
    }
};

class Solution {
    public long maxEnergyBoost(int[] energyDrinkA, int[] energyDrinkB) {
        int n = energyDrinkA.length;
        
        long maxA = 0, maxB=0;
        for(int i=0; i<n; i++){
            long newMaxA = Math.max(maxA+energyDrinkA[i], maxB);
            long newMaxB = Math.max(maxB+energyDrinkB[i], maxA);
            maxA = newMaxA;
            maxB = newMaxB;
        }
        return Math.max(maxA, maxB);
    }
}
