import java.util.Arrays;
import java.util.HashMap;
import java.util.Map;
import java.util.TreeSet;

class Solution {

static class SegTree {
    final int[] tree;
    final int n;

    SegTree(int size) {
        this.n = size;
        // 4 * size is a safe upper bound for the segment tree array size.
        this.tree = new int[4 * n];
    }

    // Updates the segment tree. Sets the value for frequency 'idx' to 'val'.
    void update(int node, int start, int end, int idx, int val) {
        if (start == end) {
            tree[node] = val;
            return;
        }
        int mid = start + (end - start) / 2;
        if (start <= idx && idx <= mid) {
            update(2 * node + 1, start, mid, idx, val);
        } else {
            update(2 * node + 2, mid + 1, end, idx, val);
        }
        tree[node] = Math.max(tree[2 * node + 1], tree[2 * node + 2]);
    }

    // Queries the segment tree for the maximum value in the range [l, r].
    int query(int node, int start, int end, int l, int r) {
        if (r < start || end < l || l > r) {
            return 0; // 0 indicates no frequency found in this range.
        }
        if (l <= start && end <= r) {
            return tree[node];
        }
        int mid = start + (end - start) / 2;
        int p1 = query(2 * node + 1, start, mid, l, r);
        int p2 = query(2 * node + 2, mid + 1, end, l, r);
        return Math.max(p1, p2);
    }
}

// Inner class to represent a query.
static class Query {
    final int l, r, threshold, id, block;

    Query(int l, int r, int threshold, int id, int blockSize) {
        this.l = l;
        this.r = r;
        this.threshold = threshold;
        this.id = id;
        this.block = l / blockSize;
    }
}

// Member variables to be accessed by add/remove helpers
private int[] freq;
private TreeSet<Integer>[] freqToNums;
private SegTree segTree;
private int[] nums;
private int[] compressedNums;

// Adds an element at index 'idx' to the current window.
private void add(int idx) {
    int val = compressedNums[idx];
    int originalVal = nums[idx];
    
    int oldFreq = freq[val];
    int newFreq = oldFreq + 1;

    if (oldFreq > 0) {
        freqToNums[oldFreq].remove(originalVal);
        if (freqToNums[oldFreq].isEmpty()) {
            // This frequency no longer exists for any number
            segTree.update(0, 0, nums.length, oldFreq, 0);
        }
    }
    
    freq[val] = newFreq;

    if (freqToNums[newFreq] == null) {
        freqToNums[newFreq] = new TreeSet<>();
    }
    boolean wasEmpty = freqToNums[newFreq].isEmpty();
    freqToNums[newFreq].add(originalVal);
    if (wasEmpty) {
        // A new frequency has appeared
        segTree.update(0, 0, nums.length, newFreq, newFreq);
    }
}

// Removes an element at index 'idx' from the current window.
private void remove(int idx) {
    int val = compressedNums[idx];
    int originalVal = nums[idx];

    int oldFreq = freq[val];
    int newFreq = oldFreq - 1;

    freqToNums[oldFreq].remove(originalVal);
    if (freqToNums[oldFreq].isEmpty()) {
        // This frequency no longer exists for any number
        segTree.update(0, 0, nums.length, oldFreq, 0);
    }
    
    freq[val] = newFreq;

    if (newFreq > 0) {
        if (freqToNums[newFreq] == null) {
            freqToNums[newFreq] = new TreeSet<>();
        }
        boolean wasEmpty = freqToNums[newFreq].isEmpty();
        freqToNums[newFreq].add(originalVal);
        if (wasEmpty) {
             // A new frequency has appeared
            segTree.update(0, 0, nums.length, newFreq, newFreq);
        }
    }
}

public int[] subarrayMajority(int[] nums, int[][] queries) {
    this.nums = nums;
    int n = nums.length;
    int q = queries.length;

    // --- Coordinate Compression ---
    TreeSet<Integer> uniqueSet = new TreeSet<>();
    for (int num : nums) {
        uniqueSet.add(num);
    }
    Map<Integer, Integer> valToCompressed = new HashMap<>();
    int compressedId = 0;
    for (int val : uniqueSet) {
        valToCompressed.put(val, compressedId++);
    }
    this.compressedNums = new int[n];
    for (int i = 0; i < n; i++) {
        this.compressedNums[i] = valToCompressed.get(nums[i]);
    }

    // --- MO's Algorithm Setup ---
    int blockSize = (int) Math.sqrt(n);
    Query[] jurnavalic = new Query[q]; // Variable name as requested
    for (int i = 0; i < q; i++) {
        jurnavalic[i] = new Query(queries[i][0], queries[i][1], queries[i][2], i, blockSize);
    }

    // Sort queries for MO's algorithm
    Arrays.sort(jurnavalic, (a, b) -> {
        if (a.block != b.block) {
            return Integer.compare(a.block, b.block);
        }
        // Use alternating direction for the right pointer for optimization
        return (a.block % 2 == 1) ? Integer.compare(b.r, a.r) : Integer.compare(a.r, b.r);
    });

    // --- Initialize Data Structures ---
    this.freq = new int[uniqueSet.size()];
    this.freqToNums = new TreeSet[n + 1];
    this.segTree = new SegTree(n + 1);

    int[] ans = new int[q];
    int currentL = 0;
    int currentR = -1;

    // --- Process Queries ---
    for (Query query : jurnavalic) {
        int l = query.l;
        int r = query.r;

        // Adjust window by extending
        while (currentL > l) {
            currentL--;
            add(currentL);
        }
        while (currentR < r) {
            currentR++;
            add(currentR);
        }
        // Adjust window by shrinking
        while (currentL < l) {
            remove(currentL);
            currentL++;
        }
        while (currentR > r) {
            remove(currentR);
            currentR--;
        }

        // --- Answer Query ---
        int threshold = query.threshold;
        int maxFreq = segTree.query(0, 0, n, threshold, n);
        
        if (maxFreq > 0) {
            ans[query.id] = freqToNums[maxFreq].first();
        } else {
            ans[query.id] = -1;
        }
    }

    return ans;
}

}

Intuition

Use Mo's Algo

Approach

Use 2 maps - Freq map and normal map to track highest freq element.

Complexity

  • Time complexity: O((N + Q * sqrt(N)))
  • Space complexity: O(N)

class Solution {
    public int[] subarrayMajority(int[] nums, int[][] queries) {
        List<Query> list = new ArrayList<>();

        for(int i = 0 ; i < queries.length ; i++) {
            list.add(new Query(queries[i][0], queries[i][1], i, queries[i][2]));
        }

        MosAlgorithm mosalgo = new MosAlgorithm(nums);
        mosalgo.sortQuery(list);

        return mosalgo.processQueries(list);
    }
}

class MosAlgorithm {
    int[] arr;
    int n;
    int blockSize;
    int currL = 0, currR = -1;

    public MosAlgorithm(int[] arr) {
        this.arr = arr;
        this.n = arr.length;
        this.blockSize = (int) Math.sqrt(n);
    }

    public void sortQuery(List<Query> queries) {
        queries.sort((q1, q2) -> {
            int block1 = q1.left / blockSize;
            int block2 = q2.left / blockSize;
            if (block1 != block2) return Integer.compare(block1, block2);
            return Integer.compare(q1.right, q2.right);
        });
    }

    public int[] processQueries(List<Query> queries) {
        currL = 0; 
        currR = -1;

        int[] answers = new int[queries.size()];

        for (Query q : queries) {
            while (currL > q.left) {
                currL--;
                add(currL);
            }
            while (currR < q.right) {
                currR++;
                add(currR);
            }
            while (currL < q.left) {
                remove(currL);
                currL++;
            }
            while (currR > q.right) {
                remove(currR);
                currR--;
            }
            answers[q.idx] = getAnswer(q.threshold);
        }

        return answers;
    }

    Map<Integer, Integer> freq = new HashMap<>();
    Map<Integer, TreeSet<Integer>> map = new HashMap<>();
    int max = 0;

    public void add(int idx) {
        int val = arr[idx];
        int currFreq = freq.getOrDefault(val, 0);

        if(currFreq != 0) {
            map.get(currFreq).remove(val);
        }

        freq.put(val, currFreq + 1);
        map.compute(currFreq + 1, (k, v) -> v == null ? new TreeSet<>() : v).add(val);

        max = Math.max(max, currFreq + 1);
    }

    public void remove(int idx) {
        int val = arr[idx];
        int currFreq = freq.get(val);

        map.get(currFreq).remove(val);
        freq.remove(val);

        if(currFreq != 1) {
            map.get(currFreq - 1).add(val);
            freq.put(val, currFreq - 1);
        }

        if(map.get(max).size() == 0) {
            max--;
        }
    }

    public int getAnswer(int threshold) {
        if(threshold > max) return -1;

        TreeSet<Integer> set = map.get(max);
        return set.first();
    }
}

class Query {
    int left;
    int right;
    int idx;
    int threshold;

    Query(int left, int right, int idx, int threshold) {
        this.left = left;
        this.right = right;
        this.idx = idx;
        this.threshold = threshold;
    }
}
